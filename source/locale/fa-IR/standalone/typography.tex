\cs_new_protected:Npn \olsa_typography_display:n #1
 {
  \prop_get:NVN \g_olsa_typography_punctuation_prop
    \l_olsa_typography_rule_tl \l_olsa_typography_punctuation_tl
  \str_case:VnF \l_olsa_typography_punctuation_tl
   {
    {none}{\[#1\]}
    % A terminal sentence mark must not become a one-character prose line.
    % Consume exactly the original following mark and retain it in \text.
    {period}{\peek_meaning_remove:NTF .
      {\[#1\text{.}\]}
      {\PackageError{standalone-typography}{Expected~terminal~period}{The~pinned~source~punctuation~changed.}\[#1\]}}
    {comma}{\peek_meaning_remove:NTF ،
      {\[#1\text{،}\]}
      {\PackageError{standalone-typography}{Expected~Persian~comma}{The~pinned~source~punctuation~changed.}\[#1\]}}
    % Keep a long source formula inline while forbidding a line break inside
    % it. The captured mathematical payload is emitted without token changes.
    {atomic}{\mbox{\c_math_toggle_token #1\c_math_toggle_token}}
   }
   {\PackageError{standalone-typography}{Unknown~typography~policy}{Check~the~exact~formula~registry.}\[#1\]}
 }
\NewDocumentCommand \OLSAProcessInlineMath {+m}
 {
  \olsa_typography_normalize:n{#1}
  \tl_set:Nx \l_olsa_typography_key_tl
    {\OLStandaloneCurrentID|\l_olsa_typography_normal_tl}
  \prop_get:NVNTF \g_olsa_typography_formula_prop
    \l_olsa_typography_key_tl \l_olsa_typography_rule_tl
   {
    \prop_get:NVNTF \g_olsa_typography_seen_prop
      \l_olsa_typography_rule_tl \l_olsa_typography_count_tl
      {}{\tl_set:Nn\l_olsa_typography_count_tl{0}}
    \tl_set:Nx \l_olsa_typography_count_tl
      {\int_eval:n{\l_olsa_typography_count_tl+1}}
    \prop_gput:NVV \g_olsa_typography_seen_prop
      \l_olsa_typography_rule_tl \l_olsa_typography_count_tl
    \prop_get:NVN \g_olsa_typography_expected_prop
      \l_olsa_typography_rule_tl \l_olsa_typography_expected_tl
    \int_compare:nNnT {\l_olsa_typography_count_tl}>{\l_olsa_typography_expected_tl}
      {\PackageError{standalone-typography}{Excess~display~rule~\l_olsa_typography_rule_tl}{Each~pinned~formula~must~have~its~exact~declared~occurrence~count.}}
    \iow_term:x{OL-STANDALONE-TYPOGRAPHY|\OLStandaloneCurrentID|\l_olsa_typography_rule_tl|\l_olsa_typography_count_tl}
    \iow_now:Nx\g_olsa_typography_iow{\OLStandaloneCurrentID|\l_olsa_typography_rule_tl|\l_olsa_typography_count_tl}
    \olsa_typography_display:n{#1}
   }
   {
    % Primitive math-shift tokens, never another active-dollar invocation.
    % This also preserves nested inline math in proof-tree boxes and \text.
    \c_math_toggle_token #1\c_math_toggle_token
   }
 }

  \iow_close:N\g_olsa_typography_iow
 }
