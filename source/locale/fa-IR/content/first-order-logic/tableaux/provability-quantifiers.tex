\olsection{\usetoken{S}{derivability} و سورها}

\begin{explain}
  قضیهٔ تمامیت همچنین ایجاب می‌کند که قواعد تابلو واقعیت‌های مربوط
  به~$\Proves$ را که در این بخش اثبات می‌شوند نتیجه دهند.
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} اگر $c$ ثابتی باشد که در $\Gamma$
یا~$!A(x)$ رخ نمی‌دهد و $\Gamma \Proves !A(c)$، آنگاه $\Gamma
\Proves \lforall[x][!A(x)]$.
\end{thm}

\begin{proof}
فرض کنید $\Gamma \Proves !A(c)$؛ یعنی $!B_1$، \dots، $!B_n
\in \Gamma$ و !!{tableau} بسته‌ای برای مجموعهٔ زیر وجود دارند:
\begin{align*}
\{ \sFmla{\False}{!A(c)}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\intertext{باید نشان دهیم برای مجموعهٔ زیر نیز !!{tableau} بسته‌ای وجود دارد:}
\{ \sFmla{\False}{\lforall[x][!A(x)]}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\end{align*}
!!{tableau} بسته را بگیرید و فرض نخستش را با
$\sFmla{\False}{\lforall[x][!A(x)]}$ جایگزین کنید، و
$\sFmla{\False}{!A(c)}$ را پس از فرض‌ها درج کنید.
\begin{center}
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\formula{A}(c)}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
      [\sFmla{\True}{\formula{B}_n},
        [][][]]]]]
\end{tableau}\qquad
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\lforall[x][\formula{A}(x)]}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
        [\sFmla{\True}{\formula{B}_n},
          [\sFmla{\False}{\formula{A}(c)}
        [][][]]]]]]
\end{tableau}
\end{center}
تابلو همچنان بسته است، زیرا همهٔ !!{sentence}s که پیش‌تر به‌عنوان فرض در
دسترس بودند، هنوز در بالای !!{tableau} در دسترس‌اند. سطر درج‌شده حاصل
کاربرد درست $\TRule{\False}{\lforall}$ است، زیرا !!{constant}~$c$ در
$!B_1$، \dots، $!B_n$ یا $\lforall[x][!A(x)]$ رخ نمی‌دهد؛ یعنی بالای
سطر درج‌شده در !!{tableau} تازه رخ نمی‌دهد.
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}
\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{!!{tableau} بسته‌ای برای
  $\sFmla{\False}{\lexists[x][!A(x)]}, \sFmla{\True}{!A(t)}$ چنین است:
  \begin{oltableau}
    [\sFmla{\False}{\lexists[x][\formula{A}(x)]}, just = \TAss
      [\sFmla{\True}{\formula{A}(t)}, just = \TAss
        [\sFmla{\False}{\formula{A}(t)}, just = {\TRule{\False}{\lexists}[1]}, close]]]
    \end{oltableau}}{}
\tagitem{prvAll}{!!{tableau} بسته‌ای برای
  $\sFmla{\False}{\formula{A}(t)}, \sFmla{\True}{\lforall[x][!A(x)]}$ چنین است:
  \begin{oltableau}
    [\sFmla{\False}{\formula{A}(t)}, just = \TAss
      [\sFmla{\True}{\lforall[x][\formula{A}(x)]}, just = \TAss
        [\sFmla{\True}{\formula{A}(t)}, just = {\TRule{\True}{\lforall}[2]}, close]]]
    \end{oltableau}}{}
\end{tagenumerate}
\end{proof}
