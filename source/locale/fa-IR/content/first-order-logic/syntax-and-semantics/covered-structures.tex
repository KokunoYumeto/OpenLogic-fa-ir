\olsection{\printtoken{P}{structure} پوشیده برای زبان‌های مرتبهٔ اول}

\begin{explain}
به یاد آورید که یک ترم هنگامی \emph{بسته} است که هیچ !!{variable}ای
در بر نداشته باشد.
\end{explain}

\begin{defn}[!!^{value} ترم‌های بسته]
اگر $t$ ترمی بسته از زبان~$\Lang L$ و $\Struct M$ یک !!{structure}
برای~$\Lang L$ باشد، \emph{!!{value}}~$\Value{t}{M}$ به شرح زیر
تعریف می‌شود:
\begin{enumerate}
\item اگر $t$ صرفاً !!{constant}~$c$ باشد، آنگاه $\Value{c}{M} = \Assign{c}{M}$.
\item اگر $t$ به‌صورت $\Atom{f}{t_1, \ldots, t_n}$ باشد، آنگاه
  \[
  \Value{t}{M} = \Assign{f}{M}(\Value{t_1}{M}, \ldots,
  \Value{t_n}{M}).
  \]
\end{enumerate}
\end{defn}

\begin{defn}[!!{structure} پوشیده]
یک !!{structure} \emph{پوشیده} است اگر هر عنصر دامنه !!{value} یک
ترم بسته باشد.
\end{defn}

\begin{ex}
فرض کنید ~$\Lang L$ زبانی با !!{constant}s $\Obj{zero}$،
$\Obj{one}$، $\Obj{two}$، \dots، !!{predicate} دوموضعی~$<$ و
!!{function}s دوموضعی $+$ و $\times$ باشد. یک !!a{structure}~$\Struct
M$ برای~$\Lang L$ را با دامنهٔ $\Domain M = \{0, 1, 2, \ldots
\}$ و انتساب‌های $\Assign{\Obj{zero}}{M} = 0$،
$\Assign{\Obj{one}}{M} = 1$، $\Assign{\Obj{two}}{M} = 2$ و به همین
ترتیب در نظر بگیرید. برای نماد رابطهٔ دوموضعی $<$، مجموعهٔ
$\Assign{<}{M}$ مجموعهٔ همهٔ زوج‌های
$\tuple{c_1, c_2} \in \Domain{M}^2$ است که در آن‌ها $c_1$ از~$c_2$
کوچک‌تر است: برای مثال، $\tuple{1, 3} \in \Assign{<}{M}$، اما
$\tuple{2, 2} \notin \Assign{<}{M}$. برای !!{function} دوموضعی $+$،
$\Assign{+}{M}$ را به شیوهٔ معمول تعریف کنید---برای مثال،
$\Assign{+}{M}$ زوج $(2,3)$ را به~$5$ نگاشت می‌کند؛ و برای
!!{function} دوموضعی~$\times$ نیز به همین ترتیب. ازاین‌رو، !!{value}
$\Obj{four}$ صرفاً~$4$ است و !!{value} عبارت
$\times(\Obj{two}, +(\Obj{three},\Obj{zero}))$ (یا در نمادگذاری
میانوندی، $\Obj{two} \times (\Obj{three} + \Obj{zero})$) چنین است:
\begin{multline*}
\Value{\times(\Obj{two}, +(\Obj{three},\Obj{zero}))}{M} =\\
\begin{aligned}
&\Assign{\times}{M}(\Value{\Obj{two}}{M}, \Value{+(\Obj{three}, \Obj{zero})}{M})\\
& = \Assign{\times}{M}(\Value{\Obj{two}}{M}, \Assign{+}{M}(\Value{\Obj{three}}{M},
\Value{\Obj{zero}}{M})) \\
& = \Assign{\times}{M}(\Assign{\Obj{two}}{M}, \Assign{+}{M}(\Assign{\Obj{three}}{M},
\Assign{\Obj{zero}}{M})) \\
& = \Assign{\times}{M}(2, \Assign{+}{M}(3, 0)) \\
& = \Assign{\times}{M}(2, 3) \\
& = 6
\end{aligned}
\end{multline*}
\end{ex}

\begin{prob}
آیا $\Struct N$، یعنی مدل استاندارد حساب، پوشیده است؟ توضیح دهید.
\end{prob}
