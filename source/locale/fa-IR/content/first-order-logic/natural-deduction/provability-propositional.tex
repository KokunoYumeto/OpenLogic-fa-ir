\olsection{\usetoken{S}{derivability} و رابط‌های گزاره‌ای}

\begin{explain}
  ثابت می‌کنیم رابطهٔ !!{derivability}~$\Proves$ در استنتاج طبیعی
  به‌اندازه‌ای قوی است که برخی واقعیت‌های بنیادیِ مربوط به رابط‌های
  گزاره‌ای، مانند $!A \land !B \Proves !A$ و
  $!A, !A \lif !B \Proves !B$ (وضع مقدم)، را اثبات کند. این واقعیت‌ها
  برای اثبات قضیهٔ تمامیت لازم‌اند.
\end{explain}

\begin{prop}\ollabel{prop:provability-land}
  \begin{enumerate}
  \item \ollabel{prop:provability-land-left} هم $!A \land !B \Proves
    !A$ و هم $!A \land !B \Proves !B$.
  \item \ollabel{prop:provability-land-right} $!A, !B \Proves !A \land !B$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item هر دو مورد زیر را می‌توانیم !!{derive} کنیم:
    \begin{prooftree}
      \AxiomC{$!A \land !B$}
      \RightLabel{\Elim{\land}}
      \UnaryInfC{$!A$}
      \DisplayProof\qquad\bottomAlignProof
      \AxiomC{$!A \land !B$}
      \RightLabel{\Elim{\land}}
      \UnaryInfC{$!B$}
    \end{prooftree}
  \item عبارت زیر را می‌توانیم !!{derive} کنیم:
    \begin{prooftree}
      \AxiomC{$!A$}
      \AxiomC{$!B$}
      \RightLabel{\Intro{\land}}
      \BinaryInfC{$!A \land !B$}
    \end{prooftree}
  \end{enumerate}
\end{proof}


\begin{prop}\ollabel{prop:provability-lor}
  \begin{enumerate}
  \item $!A \lor !B, \lnot !A, \lnot !B$ ناسازگار است.
  \item هم $!A \Proves !A \lor !B$ و هم $!B \Proves !A \lor !B$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item !!{derivation} زیر را در نظر بگیرید:
    \begin{prooftree}
      \AxiomC{$!A \lor !B$}
      \AxiomC{$\lnot !A$}
      \AxiomC{$\Discharge{!A}{1}$}
      \RightLabel{\Elim{\lnot}}
      \BinaryInfC{$\lfalse$}
      \AxiomC{$\lnot !B$}
      \AxiomC{$\Discharge{!B}{1}$}
      \RightLabel{\Elim{\lnot}}
      \BinaryInfC{$\lfalse$}
      \DischargeRule{\Elim{\lor}}{1}
      \TrinaryInfC{$\lfalse$}
    \end{prooftree}
    این !!a{derivation} از~$\lfalse$ از فرض‌های !!{undischarged}
    $!A \lor !B$، $\lnot !A$ و $\lnot !B$ است.
  \item هر دو مورد زیر را می‌توانیم !!{derive} کنیم:
    \begin{prooftree}
      \AxiomC{$!A$}
      \RightLabel{\Intro{\lor}}
      \UnaryInfC{$!A \lor !B$}
      \DisplayProof\qquad\bottomAlignProof
      \AxiomC{$!B$}
      \RightLabel{\Intro{\lor}}
      \UnaryInfC{$!A \lor !B$}
    \end{prooftree}
  \end{enumerate}
\end{proof}

\begin{prop}\ollabel{prop:provability-lif}
  \begin{enumerate}
  \item \ollabel{prop:provability-lif-left}  $!A, !A \lif !B \Proves !B$.
  \item \ollabel{prop:provability-lif-right}
    هم $\lnot !A \Proves !A \lif !B$ و هم $!B \Proves !A \lif !B$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item عبارت زیر را می‌توانیم !!{derive} کنیم:
    \begin{prooftree}
      \AxiomC{$!A \lif !B$}
      \AxiomC{$!A$}
      \RightLabel{\Elim{\lif}}
      \BinaryInfC{$!B$}
    \end{prooftree}

  \item این مطلب را دو !!{derivation}s زیر نشان می‌دهند:
    \begin{prooftree}
      \AxiomC{$\lnot !A$}
      \AxiomC{$\Discharge{!A}{1}$}
      \RightLabel{\Elim{\lnot}}
      \BinaryInfC{$\lfalse$}
      \RightLabel{\FalseInt}
      \UnaryInfC{$!B$}
      \DischargeRule{\Intro{\lif}}{1}
      \UnaryInfC{$!A \lif !B$}
      \DisplayProof\qquad\bottomAlignProof
      \AxiomC{$!B$}
      \RightLabel{\Intro{\lif}}
      \UnaryInfC{$!A \lif !B$}
    \end{prooftree}
    توجه کنید که $\Intro{\lif}$ می‌تواند فرض~$!A$ را !!{discharge}
    کند، اما الزامی به این کار ندارد.
  \end{enumerate}
\end{proof}
