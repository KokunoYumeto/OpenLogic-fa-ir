\section{مدل‌های $\Th{Q}$}

\begin{explain}
می‌دانیم !!{structure}s نااستانداردی وجود دارند که همان !!{sentence}sی
را صادق می‌کنند که~$\Struct{N}$ صادق می‌کند؛ یعنی مدلِ~$\Th{TA}$
هستند. چون $\Sat{N}{\Th{Q}}$، هر مدل~$\Th{TA}$ مدل~$\Th{Q}$ نیز
هست. $\Th{Q}$ بسیار ضعیف‌تر از~$\Th{TA}$ است؛ برای نمونه،
$\Th{Q} \Proves/ \lforall[x][\lforall[y][\eq[(x +
      y)][(y+x)]]]$. ارضاکردن نظریه‌های ضعیف‌تر آسان‌تر است: مدل‌های
بیشتری دارند. برای نمونه، $\Th{Q}$ مدل‌هایی دارد که
$\lforall[x][\lforall[y][\eq[(x + y)][(y+x)]]]$ را کاذب می‌کنند، اما
این مدل‌ها نمی‌توانند مدل~$\Th{TA}$ یا، به طریق اولی، مدل~$\Th{PA}$
نیز باشند. مدل‌های~$\Th{Q}$ همچنین نسبتاً ساده‌اند: می‌توانیم آن‌ها را
به‌صراحت مشخص کنیم.
\end{explain}

\begin{ex}
\ollabel{ex:model-K-of-Q}
!!{structure}~$\Struct{K}$ را با دامنهٔ
$\Domain{K} = \Nat \cup \{a\}$ و تعبیرهای زیر در نظر بگیرید:
\begin{align*}
  \Assign{\Obj{0}}{K} & = 0\\
  \Assign{\prime}{K}(x) & =
  \begin{cases}
    x+1 & \text{اگر $x\in \Nat$}\\
    a & \text{اگر $x = a$}
  \end{cases}\\
  \Assign{+}{K}(x, y) & =
  \begin{cases}
    x+y & \text{اگر $x$، $y \in\Nat$}\\
    a & \text{در غیر این صورت}
  \end{cases}\\
  \Assign{\times}{K}(x, y) & =
  \begin{cases}
    xy & \text{اگر $x$، $y \in\Nat$}\\
    0 & \text{اگر $x = 0$ یا $y = 0$}\\
    a & \text{در غیر این صورت}\\
  \end{cases}\\
  \Assign{<}{K} & =
  \Setabs{\tuple{x,y}}{x, y \in \Nat \text{ و } x<y} \cup
  \Setabs{\tuple{x,a}}{x \in \Domain{K}}.
\end{align*}
برای نشان‌دادن $\Sat{K}{\Th{Q}}$ باید وارسی کنیم که همهٔ اصول
$\Th{Q}$ در~$\Struct{K}$ صادق‌اند. برای سهولت،
$x^\nssucc$ را به‌جای $\Assign{\prime}{K}(x)$ («جانشین»~$x$
در~$\Struct{K}$)، $x \nsplus y$ را به‌جای $\Assign{+}{K}(x, y)$
(«مجموع»~$x$ و~$y$ در~$\Struct{K}$)، $x \nstimes y$ را به‌جای
$\Assign{\times}{K}(x, y)$ («حاصل‌ضرب»~$x$ و~$y$ در~$\Struct{K}$)،
و $x \nsless y$ را به‌جای
$\tuple{x,y} \in \Assign{<}{K}$ می‌نویسیم. با این اختصارها می‌توانیم
عملیات~$\Struct{K}$ را روشن‌تر به صورت زیر نمایش دهیم:
\[
\begin{array}{c|c}
  x & x^\nssucc \\
  \hline
  n & n+1 \\
  a & a
\end{array}
\qquad
\begin{array}{c|ccc}
  x \nsplus y & 0 & m & a \\
  \hline
  0 & 0 & m & a \\
  n & n & n+m & a \\
  a & a & a & a \\
\end{array}
\qquad
\begin{array}{c|ccc}
  x \nstimes y & 0 & m & a \\
  \hline
  0 & 0 & 0 & 0 \\
  n & 0 & nm & a \\
  a & 0 & a & a \\
\end{array}
\]
برای $n$، $m \in \Nat$، $n \nsless m$ هرگاه و تنها هرگاه $n<m$؛
همچنین برای هر $x \in \Domain{K}$، $x \nsless a$.

چون $\nssucc$ !!{injective} است،
$\Sat{K}{\lforall[x][\lforall[y][(\eq[x'][y'] \lif \eq[x][y])]]}$.
چون~$0$ در~$\Struct{K}$ جانشینِ~$\nssucc$ نیست،
$\Sat{K}{\lforall[x][\eq/[\Obj 0][x']]}$. چون برای هر $n>0$،
$n = (n-1)^\nssucc$ و $a = a^\nssucc$،
$\Sat{K}{\lforall[x][(\eq[x][\Obj 0] \lor
\lexists[y][\eq[x][y']])]}$.

چون $n \nsplus 0 = n+0 = n$ و بنا بر تعریف~$\nsplus$،
$a\nsplus 0 = a$، داریم
$\Sat{K}{\lforall[x][\eq[(x + \Obj 0)][x]]}$. نشان‌دادن
$\Sat{K}{\lforall[x][\lforall[y][\eq[(x + y')][(x + y)']]]}$ اندکی
دشوارتر است. اگر $n$ و~$m$ هر دو استاندارد باشند، داریم:
\begin{align*}
(n \nsplus m^\nssucc) & = (n+(m+1)) = (n+m)+1 = (n \nsplus m)^\nssucc
\intertext{زیرا $\nsplus$ و $^\nssucc$ روی اعداد استاندارد با $+$ و
  $\prime$ توافق دارند. اکنون فرض کنید $x \in \Domain{K}$. آنگاه}
(x \nsplus a^\nssucc) & = (x \nsplus a) = a = a^\nssucc = (x \nsplus a)^\nssucc.
\intertext{حالت باقی‌مانده آن است که $y \in \Domain{K}$ اما $x = a$.
  اینجا نیز باید برحسب اینکه $y=n$ استاندارد است یا $y=a$، دو حالت را
  جدا کنیم:}
(a \nsplus n^\nssucc) & = (a \nsplus (n+1)) = a = a^\nssucc = (a \nsplus n)^\nssucc\\
(a \nsplus a^\nssucc) & = (a \nsplus a) = a = a^\nssucc = (a \nsplus a)^\nssucc.
\end{align*}
البته این شرح تفصیلی‌تر از حد لازم است. برای نمونه، چون هرچه~$z$
باشد $a \nsplus z = a$، بی‌درنگ نتیجه می‌گیریم
$a \nsplus a^\nssucc = a$. اصول باقی‌مانده را نیز می‌توان به همین
شیوه وارسی کرد.

پس $\Struct{K}$ مدلی از~$\Th{Q}$ است. «جمع» آن، یعنی~$\nsplus$،
همچنین جابجایی است. اما جمله‌های دیگری نیز وجود دارند که
در~$\Struct{N}$ صادق و در~$\Struct{K}$ کاذب‌اند، و برعکس. برای نمونه،
$a \nsless a$؛ پس $\Sat{K}{\lexists[x][x < x]}$ و
$\Sat/{K}{\lforall[x][\lnot x<x]}$. این نشان می‌دهد که
$\Th{Q} \Proves/ \lforall[x][\lnot x < x]$.
\end{ex}

\begin{prob}
ثابت کنید $\Struct{K}$ از
\olref[mod][mar][mdq]{ex:model-K-of-Q} اصول باقی‌ماندهٔ~$\Th{Q}$ را
ارضا می‌کند:
\begin{align*}
  & \lforall[x][\eq[(x \times \Obj 0)][\Obj 0]] \tag{$!Q_6$}\\
  & \lforall[x][\lforall[y][\eq[(x \times y')][((x \times y) + x)]]] \tag{$!Q_7$}\\
  & \lforall[x][\lforall[y][(x < y \liff \lexists[z][\eq[(z' + x)][y]])]] \tag{$!Q_8$}.
\end{align*}
!!a{sentence} بیابید که تنها شامل~$\prime$، در~$\Struct{N}$ صادق و
در~$\Struct{K}$ کاذب باشد.
\end{prob}

\begin{ex}
\ollabel{ex:model-L-of-Q} !!{structure}~$\Struct{L}$ را با دامنهٔ
$\Domain{L} = \Nat \cup \{a, b\}$ و تعبیرهای
$\Assign{\prime}{L} = \nssucc$ و $\Assign{+}{L} = \nsplus$ که در
جدول‌های زیر داده شده‌اند، در نظر بگیرید:
\[
\begin{array}{c|c}
  x & x^\nssucc \\
  \hline
  n & n+1 \\
  a & a\\
  b & b
\end{array}
\qquad
\begin{array}{c|ccc}
  x \nsplus y & m & a & b\\
  \hline
  n & n+m & b & a\\
  a & a & b & a\\
  b & b & b & a
\end{array}
\]
چون $\nssucc$ !!{injective} است، $0$ در برد آن نیست و هر
$x \in \Domain{L}$ جز~$0$ در برد آن است، اصول $!Q_1$--$!Q_3$
در~$\Struct{L}$ صادق‌اند. برای هر~$x$، $x \nsplus 0 = x$؛ پس
$!Q_4$ نیز صادق است. برای~$!Q_5$، عبارت‌های
$x \nsplus y^\nssucc$ و $(x \nsplus y)^\nssucc$ را در نظر بگیرید.
اگر $x$ و~$y$ هر دو استاندارد باشند، این دو برابرند، زیرا در این صورت
$\nssucc$ و~$\nsplus$ با $\prime$ و~$+$ توافق دارند. اگر $x$
نااستاندارد و $y$ استاندارد باشد، داریم
$x \nsplus y^\nssucc = x = x^\nssucc = (x \nsplus y)^\nssucc$.
اگر $x$ و~$y$ هر دو نااستاندارد باشند، چهار حالت داریم:
\begin{align*}
&  a \nsplus a^\nssucc  = b = b^\nssucc = (a \nsplus a)^\nssucc\\
&  b \nsplus b^\nssucc  = a = a^\nssucc = (b \nsplus b)^\nssucc\\
&  b \nsplus a^\nssucc  = b = b^\nssucc = (b \nsplus a)^\nssucc\\
&  a \nsplus b^\nssucc  = a = a^\nssucc = (a \nsplus b)^\nssucc\\
\intertext{اگر $x$ استاندارد اما $y$ نااستاندارد باشد، داریم}
&  n \nsplus a^\nssucc  = n \nsplus a = b = b^\nssucc = (n \nsplus a)^\nssucc\\
&  n \nsplus b^\nssucc  = n \nsplus b = a = a^\nssucc = (n \nsplus b)^\nssucc.
\end{align*}
پس $\Sat{L}{!Q_5}$. بااین‌حال،
$a \nsplus 0 \neq 0 \nsplus a$؛ بنابراین
$\Sat/{L}{\lforall[x][\lforall[y][\eq[(x+y)][(y+x)]]]}$.
\end{ex}

\begin{prob}
$\Struct{L}$ از
\olref[mod][mar][mdq]{ex:model-L-of-Q} را با افزودن~$\nstimes$ و
$\nsless$ برای تعبیر~$\times$ و~$<$ بسط دهید. نشان دهید ساختارتان اصول
باقی‌ماندهٔ~$\Th{Q}$ را ارضا می‌کند:
\begin{align*}
& \lforall[x][\eq[(x \times \Obj 0)][\Obj 0]] \tag{$!Q_6$}\\ &
  \lforall[x][\lforall[y][\eq[(x \times y')][((x \times y) + x)]]]
  \tag{$!Q_7$}\\ & \lforall[x][\lforall[y][(x < y \liff \lexists[z][\eq[(z'+x)][y]])]] \tag{$!Q_8$}.
\end{align*}
\end{prob}

\begin{prob}
در~$\Struct{L}$ از
\olref[mod][mar][mdq]{ex:model-L-of-Q}،
$a^\nssucc = a$ و $b^\nssucc = b$. آیا مدلی از~$\Th{Q}$ وجود دارد
که در آن $a^\nssucc = b$ و $b^\nssucc = a$؟
\end{prob}

\begin{explain}
مدل‌هایی از~$\Th{Q}$ ساخته‌ایم که در آن‌ها !!{element}s نااستاندارد
«پس از» عناصر استاندارد جای دارند. در واقع، اصول همین اندازه را ایجاب
می‌کنند. !!{element} نااستاندارد چون~$x$ نمی‌تواند
${} \nsless 0$ باشد، زیرا
$\Th{Q} \Proves \lforall[x][\lnot x<0]$ (بنگرید به
\olref[inc][req][min]{lem:less-zero}). همچنین برای هر~$n$،
$\Th{Q} \Proves \lforall[x][(x < \num{n}' \lif
(\eq[x][\num{0}] \lor \eq[x][\num{1}] \lor \dots \lor
\eq[x][\num{n}]))]$ (\olref[inc][req][min]{lem:less-nsucc})؛ پس
نمی‌توانیم برای هیچ $n>0$ داشته باشیم $a \nsless n$.
\end{explain}
