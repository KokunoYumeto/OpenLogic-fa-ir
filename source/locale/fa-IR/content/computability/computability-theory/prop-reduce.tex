\olsection{ویژگی‌های کاهش‌پذیری}

اگر $A$ به~$B$ کاهش یابد می‌نویسیم $A\leq_m B$. این نمادگذاری القا
می‌کند که اگر $A\leq_m B$، آنگاه $A$ «از~$B$ دشوارتر نیست» و $B$
«هم‌اندازهٔ~$A$ یا دشوارتر از آن» است، و نیز مانند رابطهٔ معمول
$\le$ روی اعداد، رابطهٔ $\le_m$ مجموعه‌ها (یا مسائل تصمیم) را برحسب
پیچیدگی مرتب می‌کند. دو گزارهٔ زیر این شهود را تأیید می‌کنند. نخستین
می‌گوید $\le_m$ متعدی است.

\begin{prop}
\ollabel{prop:trans-red}
اگر $A\leq_m B$ و $B\leq_m C$، آنگاه $A\leq_m C$.
\end{prop}

\begin{proof}
ترکیب یک کاهش از $A$ به~$B$ با یک کاهش از $B$ به~$C$، کاهشی از $A$
به~$C$ به دست می‌دهد.
\end{proof}

\begin{prob}
\olref{prop:trans-red} را ثابت کنید: نشان دهید اگر $f$ و~$g$
به‌ترتیب کاهش‌های چندبه‌یک از $A$ به~$B$ و از $B$ به~$C$ باشند،
آنگاه $\comp{f}{g}$ کاهش چندبه‌یک از $A$ به~$C$ است.
\end{prob}

\begin{prop}
\ollabel{prop:reduce}
بگذارید $A$ و~$B$ هر دو مجموعه باشند و فرض کنید $A\leq_m B$.
\begin{enumerate}
\item اگر $B$ محاسبه‌پذیرِ برشمردنی باشد، $A$ نیز چنین است.
\item اگر $B$ محاسبه‌پذیر باشد، $A$ نیز چنین است.
\end{enumerate}
\end{prop}

\begin{proof}
بگذارید $f$ کاهش چندبه‌یک از $A$ به~$B$ باشد. برای ادعای نخست کافی
است بررسی کنید که اگر $B$ دامنهٔ تابع جزئی~$g$ باشد، آنگاه $A$
دامنهٔ~$\comp{f}{g}$ است:
\begin{align*}
x \in A & \text{ اگر و تنها اگر } f(x) \in B \\
& \text{ اگر و تنها اگر }  g(f(x)) \fdefined.
\end{align*}

برای ادعای دوم به یاد آورید که اگر $B$ محاسبه‌پذیر باشد، $B$ و
$\Complement{B}$ هر دو محاسبه‌پذیرِ برشمردنی‌اند
(\olref[cmp]{thm:ce-comp}). به‌آسانی می‌توان بررسی کرد که $f$ همچنین
کاهش چندبه‌یکِ $\Complement{A}$ به $\Complement{B}$ است؛ پس بنا بر
بخش نخست این برهان، $A$ و~$\Complement{A}$ هر دو !!{computably
enumerable} هستند. بنابراین بنا بر \olref[cmp]{thm:ce-comp}، $A$
محاسبه‌پذیر است. (یا می‌توانید بررسی کنید
$\Char{A}=\comp{f}{\Char{B}}$؛ پس اگر $\Char{B}$ محاسبه‌پذیر باشد،
$\Char{A}$ نیز چنین است.)
\end{proof}

\begin{prob}
فرض کنید $f$ کاهش چندبه‌یکِ $A$ به~$B$ باشد. نشان دهید $f$ همچنین
کاهش چندبه‌یکِ $\Complement{A}$ به $\Complement{B}$ است.
\end{prob}

\begin{prob}
نشان دهید اگر $f$ کاهش چندبه‌یکِ $A$ به~$B$ باشد، آنگاه
$\Char{A}=\comp{f}{\Char{B}}$.
\end{prob}

\begin{digress}
مفهومی کلی‌تر از کاهش‌پذیری به نام \emph{کاهش‌پذیری تورینگ} در
زمینه‌های دیگر، به‌ویژه برای اثبات نتایج تصمیم‌ناپذیری، سودمند است.
توجه کنید که بنا بر \olref[cmp]{cor:comp-k}، متمم~$K_0$ به~$K_0$
کاهش‌پذیر نیست، زیرا محاسبه‌پذیرِ برشمردنی نیست. اما به‌طور شهودی،
اگر پاسخ پرسش‌های مربوط به $K_0$ را می‌دانستید، پاسخ پرسش‌های مربوط
به متمم آن را نیز می‌دانستید. می‌گوییم مجموعهٔ $A$ به~$B$
کاهش‌پذیر تورینگ است اگر بتوان با روشی محاسبه‌پذیر که می‌تواند
دربارهٔ~$B$ پرسش کند، پاسخ پرسش‌های مربوط به~$A$ را تعیین کرد. این
مفهوم از کاهش‌پذیری چندبه‌یک آزادانه‌تر است؛ در آن (۱)~تنها مجازید یک
پرسش دربارهٔ $B$ بپرسید و (۲)~پاسخ «آری» باید به پاسخ «آری» به پرسش
مربوط به $A$ ترجمه شود و برای «نه» نیز همین‌طور. همچنان اگر $A$ به
$B$ کاهش‌پذیر تورینگ و $B$ محاسبه‌پذیر باشد، $A$ نیز محاسبه‌پذیر است
(هرچند چنان‌که دیدیم، گزارهٔ همانند برای محاسبه‌پذیریِ برشمردنی برقرار
نیست).

باید دربارهٔ مفهوم‌های گوناگون کاهش‌پذیری که بررسی کرده‌ایم بیندیشید
و تفاوت‌هایشان را بفهمید. بااین‌حال، در این فصل تنها با کاهش‌پذیری
چندبه‌یک سروکار خواهیم داشت. ضمناً هر دو نوع کاهش‌پذیریِ بند پیشین در
پیچیدگی محاسباتی همتایانی دارند، با این شرط اضافی که ماشین‌های تورینگ
در زمان چندجمله‌ای اجرا شوند: نسخهٔ پیچیدگیِ کاهش‌پذیری چندبه‌یک
\emph{کاهش‌پذیری کارپ} و نسخهٔ پیچیدگیِ کاهش‌پذیری تورینگ
\emph{کاهش‌پذیری کوک} نام دارد.
\end{digress}
