\olsection{دنباله‌ها}

مجموعهٔ توابع بازگشتی اولیه به‌طرز چشمگیری نیرومند و پایدار است. اما
پس از پروردن روشی مناسب برای کار با 
\emph{دنباله‌ها}، کارهای بیشتری نیز می‌توانیم انجام دهیم. دنباله‌های
متناهی اعداد طبیعی را به شیوهٔ زیر با اعداد طبیعی یکی می‌گیریم:
دنبالهٔ~$\langle a_0,a_1,a_2,\dots,a_k\rangle$ متناظر است با عدد
\[
p_0^{a_0+1} \cdot p_1^{a_1+1} \cdot p_2^{a_2+1} \cdot \dots \cdot
p_k^{a_k+1}.
\]
یک واحد به توان‌ها می‌افزاییم تا تضمین شود، برای نمونه، دنباله‌های
$\langle 2,7,3\rangle$ و $\langle 2,7,3,0,0\rangle$ کدهای عددی
متمایزی داشته باشند. می‌توانیم هم~$0$ و هم~$1$ را کد دنبالهٔ تهی
بگیریم؛ برای تعیین قرارداد، بگذارید~$\emptyseq$ نشان‌دهندهٔ~$0$ باشد.

این کدگذاری دنباله‌ها به سبب آنچه قضیهٔ اساسی حساب نامیده می‌شود کار
می‌کند: هر عدد طبیعی $n \ge 2$ را می‌توان به یک و تنها یک شیوه به
صورت
\[
n = p_0^{a_0} \cdot p_1^{a_1} \cdot \dots \cdot p_k^{a_k}
\]
با $a_k \ge 1$ نوشت. این امر تضمین می‌کند نگاشت
$(a_0,\dots,a_k) \longmapsto \tuple{a_0,\dots,a_k}$ یک‌به‌یک باشد:
دنباله‌های متفاوت به اعداد متفاوت نگاشته می‌شوند و با هر عدد حداکثر
یک دنباله متناظر است.

اکنون نشان می‌دهیم عملیات تعیین طول یک دنباله، تعیین عنصر~$i$ام آن،
افزودن عنصری به انتهای دنباله و الحاق دو دنباله همگی بازگشتی اولیه‌اند.

\begin{prop}
  تابع~$\len{s}$ که طول دنبالهٔ~$s$ را بازمی‌گرداند، بازگشتی اولیه
  است.
\end{prop}

\begin{proof}
  بگذارید $R(i,s)$ رابطه‌ای باشد که با
  \[
  R(i, s) \text{ هرگاه و تنها هرگاه }
  p_i \mid s \land p_{i+1} \nmid s
  \]
  تعریف می‌شود. آشکار است که~$R$ بازگشتی اولیه است. هرگاه~$s$ کد یک
  دنبالهٔ ناتهی باشد، یعنی
  \[
  s = p_0^{a_0+1} \cdot \dots \cdot p_{k}^{a_{k}+1},
  \]
  رابطهٔ $R(i,s)$ هنگامی برقرار است که~$p_i$ بزرگ‌ترین عدد اولی باشد
  که $p_i \mid s$؛ یعنی $i=k$. پس طول~$s$ برابر~$i+1$ است هرگاه و
  تنها هرگاه~$p_i$ بزرگ‌ترین عدد اولی باشد که~$s$ را عاد می‌کند.
  بنابراین می‌توانیم قرار دهیم
  \[
  \len{s} =
  \begin{cases}
    0 & \text{اگر $s = 0$ یا $s = 1$} \\
    1 + \bmin{i < s}{R(i, s)} & \text{در غیر این صورت}.
  \end{cases}
  \]
  در اینجا می‌توانیم از کمینه‌سازی کراندار استفاده کنیم، زیرا وقتی
  $s$ کد دنباله‌ای باشد تنها یک~$i$ رابطهٔ $R(i,s)$ را برآورده
  می‌کند و، اگر چنین~$i$ای وجود داشته باشد، از خود~$s$ کوچک‌تر است.
\end{proof}

\begin{prop}
  تابع~$\fn{append}(s,a)$ که حاصل افزودن~$a$ به انتهای دنبالهٔ~$s$
  را بازمی‌گرداند، بازگشتی اولیه است.
\end{prop}

\begin{proof}
  $\fn{append}$ را می‌توان چنین تعریف کرد:
  \[
  \fn{append}(s,a) =
  \begin{cases}
    2^{a+1} & \text{اگر $s = 0$ یا $s = 1$} \\
    s \cdot p_{\len{s}}^{a+1} & \text{در غیر این صورت}.
  \end{cases}
  \]
\end{proof}

\begin{prop}
  تابع~$\fn{element}(s,i)$ که عنصر~$i$ام~$s$ را بازمی‌گرداند
  (عنصر آغازین، عنصر~$0$ام نامیده می‌شود)، و اگر $i$ بزرگ‌تر یا مساوی
  طول~$s$ باشد~$0$ را بازمی‌گرداند، بازگشتی اولیه است.
\end{prop}

\begin{proof}
توجه کنید $a$ عنصر~$i$ام~$s$ است هرگاه و تنها هرگاه
$p_i^{a+1}$ بزرگ‌ترین توان~$p_i$ باشد که~$s$ را عاد می‌کند؛ یعنی
$p_i^{a+1} \mid s$ اما $p_i^{a+2} \nmid s$. پس:
  \[
  \fn{element}(s,i) =
  \begin{cases}
    0 & \mbox{اگر $i \geq \len{s}$} \\
    \bmin{a < s}{(p_i^{a+2} \nmid s)} & \text{در غیر این صورت}.
  \end{cases}
  \]
\end{proof}

به‌جای نام‌های رسمی توابعی که در بالا تعریف شدند، نمادگذاری
فشرده‌تری معرفی می‌کنیم. به‌جای~$\fn{element}(s,i)$ از~$(s)_i$ استفاده
می‌کنیم و $\tuple{s_0,\dots,s_k}$ را اختصاری برای
\[
\fn{append}(\fn{append}(\dots \fn{append}(\emptyseq,s_0)
\dots),s_k)
\]
می‌گیریم. توجه کنید اگر~$s$ طول~$k$ داشته باشد، عناصرش عبارت‌اند از
$(s)_0$، \dots، $(s)_{k-1}$.

\begin{prop}
تابع~$\fn{concat}(s,t)$ که دو دنباله را به هم الحاق می‌کند، بازگشتی
اولیه است.
\end{prop}

\begin{proof}
  تابعی چون~$\fn{concat}$ با ویژگی زیر می‌خواهیم:
  \[
    \fn{concat}(\tuple{a_0, \dots, a_k}, \tuple{b_0, \dots, b_l}) =
    \tuple{a_0, \dots, a_k, b_0, \dots, b_l}.
  \]
  از تابع «کمکی»~$\fn{hconcat}(s,t,n)$ استفاده می‌کنیم که نخستین
  $n$ نماد~$t$ را به~$s$ الحاق می‌کند. این تابع با بازگشت اولیه چنین
  تعریف می‌شود:
  \begin{align*}
    \fn{hconcat}(s,t,0) & = s\\
    \fn{hconcat}(s,t,n+1) & = \fn{append}(\fn{hconcat}(s,t,n),(t)_n)
    \intertext{سپس می‌توانیم $\fn{concat}$ را چنین تعریف کنیم:}
    \fn{concat}(s,t) & = \fn{hconcat}(s,t,\len{t}).
  \end{align*}
\end{proof}

به‌جای $\fn{concat}(s,t)$ می‌نویسیم $s \concat t$.

برای ما سودمند خواهد بود که بتوانیم کد عددی یک دنباله را برحسب طول و
بزرگ‌ترین عنصرش کران‌گذاری کنیم. فرض کنید~$s$ دنباله‌ای به طول~$k$
است که هر عنصرش از عددی چون~$x$ کوچک‌تر یا با آن برابر است. آنگاه در
تجزیهٔ اول~$s$ حداکثر~$k$ عامل اول وجود دارد که هر یک حداکثر
$p_{k-1}$ است و هر کدام حداکثر به توان~$x+1$ رسیده است. به بیان دیگر،
قرار دهید $\fn{sequenceBound}(x,0)=1$ و برای~$k>0$ تعریف کنید
\[
\fn{sequenceBound}(x,k) = p_{k-1}^{k \cdot (x+1)}.
\]
آنگاه کد عددی دنبالهٔ~$s$ که در بالا وصف شد حداکثر
$\fn{sequenceBound}(x,k)$ است.

داشتن چنین کرانی برای دنباله‌ها راهی برای تعریف توابع تازه با جست‌وجوی
کراندار در اختیارمان می‌گذارد. برای نمونه، می‌توانیم~$\fn{concat}$ را
با جست‌وجوی کراندار تعریف کنیم. تنها لازم است 
\emph{مشخصه‌ای} بازگشتی اولیه برای شیء مورد نظرمان (عدد دنبالهٔ
الحاق‌شده) و کرانی برای گسترهٔ جست‌وجو بنویسیم. عبارت زیر مناسب است:
\begin{align*}
  \fn{concat}(s,t) = {} & \bmin{v < \fn{sequenceBound}(s+t,\len{s} +
    \len{t})}{} \\
  & \quad(\len{v} = \len{s} + \len{t} \land {}\\
    & \qquad \bforall{i < \len{s}}{((v)_i = (s)_i) \land {} \\
      & \qquad \bforall{j < \len{t}}{((v)_{\len{s}+j} = (t)_j)})}.
\end{align*}

\begin{prob}
نشان دهید تابعی بازگشتی اولیه چون~$\fn{sconcat}(s)$ با ویژگی زیر وجود
دارد:
\[
\fn{sconcat}(\tuple{s_0, \dots, s_k}) = s_0 \concat \dots \concat s_k.
\]
\end{prob}

\begin{prob}
نشان دهید تابعی بازگشتی اولیه چون~$\fn{tail}(s)$ با ویژگی زیر وجود
دارد:
\begin{align*}
  \fn{tail}(\emptyseq) & = 0 \text{ و}\\
  \fn{tail}(\tuple{s_0, \dots, s_{k}}) & = \tuple{s_1, \dots, s_{k}}.
\end{align*}
\end{prob}

\begin{prop}
  \ollabel{prop:subseq}
  تابع~$\fn{subseq}(s,i,n)$ که زیردنباله‌ای به طول~$n$ از~$s$ را با
  آغاز از عنصر~$i$ام بازمی‌گرداند، بازگشتی اولیه است.
\end{prop}

\begin{proof}
  تمرین.
\end{proof}
