\olsection{تصمیم‌پذیری}
توجه کنید که اثبات قضیهٔ تمامیت برای هر~$\Gamma \Proves/ !A$ مدلی با
بی‌نهایت جهان به دست می‌دهد که گواه~$\Gamma \Entails/ !A$ است. گزارهٔ
زیر نشان می‌دهد برای اثبات~$\Entails !A$ کافی است ثابت کنیم
$\mSat{M}{!A}$ برای همهٔ مدل‌های متناهی (یعنی مدل‌هایی با مجموعهٔ
متناهی جهان‌ها).

\begin{thm}\ollabel{thm:decidability}
  اگر~$\Entails/ !A$، آنگاه مدل متناهی~$\mModel{M'}$ای وجود دارد
  چنان‌که~$\mSat/{M'}{!A}$.
\end{thm}

\begin{proof}
  چون~$\Entails/ !A$، مدل~$\mModel{M}=\tuple{W,R,V}$ و جهان
  $w_0 \in W$ای وجود دارند چنان‌که~$\mSat/{M}{!A}[w_0]$. بگذارید
  $\Sigma$ مجموعهٔ متناهی همهٔ زیر‌فرمول‌های~$!A$ و~$P$ مجموعهٔ
  !!{propositional variable}های ظاهرشده در~$!A$ باشد. روی~$W$ رابطهٔ
  هم‌ارزی زیر را تعریف می‌کنیم:
  \[
    w \sim v \quad\text{اگر و تنها اگر}\quad
    \text{برای هر~$!B \in \Sigma$، }
    \mSat{M}{!B}[w] \text{ اگر و تنها اگر } \mSat{M}{!B}[v].
  \]
  کلاس هم‌ارزی~$w$ را با~$[w]$ نشان دهید و تعریف کنید
  \[
    W' = \Setabs{[w]}{w \in W}.
  \]
  همچنین~$R'[w][v]$ را اگر و تنها اگر برای هر~$!B \in \Sigma$،
  $\mSat{M}{!B}[w]$ مستلزم~$\mSat{M}{!B}[v]$ باشد قرار دهید. برای
  $p \in P$ تعریف کنید
  \[
    V'(p) = \Setabs{[w] \in W'}{\mSat{M}{p}[w]},
  \]
  و برای~$p \notin P$ بگذارید~$V'(p)=\emptyset$.
  این تعریف‌ها به نمایندگان کلاس‌ها وابسته نیستند. همچنین~$W'$ متناهی
  است، زیرا هر کلاس با مجموعهٔ فرمول‌های~$\Sigma$ که در آن صادق‌اند
  تعیین می‌شود؛ پس~$|W'| \le 2^{|\Sigma|}$. رابطهٔ~$R'$ بازتابی و تعدی
  است. اگر هم~$R'[w][v]$ و هم~$R'[v][w]$، آنگاه~$w \sim v$، پس
  $[w]=[v]$؛ بنابراین~$R'$ پادتقارنی نیز هست. تعریف~$R'$ همچنین
  یکنوایی~$V'$ را تضمین می‌کند. در نتیجه
  $\mModel{M'}=\tuple{W',R',V'}$ یک !!a{relational model} متناهی است.

  اکنون لم تصفیه را با استقرا بر ساختار فرمول ثابت می‌کنیم: برای هر
  !!{formula}~$!B \in \Sigma$ و هر~$w \in W$،
  \[
    \mSat{M}{!B}[w] \quad\text{اگر و تنها اگر}\quad
    \mSat{M'}{!B}[[w]].
  \]
  حالت‌های~$\lfalse$ و هر !!{propositional variable}~$p \in P$ از
  تعریف~$V'$ نتیجه می‌شوند، و حالت‌های~$\land$ و~$\lor$ بی‌درنگ از فرض
  استقرا به دست می‌آیند. برای شرطی، بگذارید~$!B$ برابر
  $!C \lif !D$ باشد. نخست فرض کنید
  $\mSat{M}{!C \lif !D}[w]$، $R'[w][v]$ و
  $\mSat{M'}{!C}[[v]]$. بنا بر تعریف~$R'$،
  $\mSat{M}{!C \lif !D}[v]$، و بنا بر فرض استقرا
  $\mSat{M}{!C}[v]$. بازتابی‌بودن~$R$ اکنون
  $\mSat{M}{!D}[v]$ را به دست می‌دهد، و فرض استقرا نتیجه می‌دهد
  $\mSat{M'}{!D}[[v]]$. پس
  $\mSat{M'}{!C \lif !D}[[w]]$.

  برعکس، اگر~$\mSat/{M}{!C \lif !D}[w]$، جهانی~$v$ با~$Rwv$ وجود
  دارد چنان‌که~$\mSat{M}{!C}[v]$ و~$\mSat/{M}{!D}[v]$. یکنوایی صدق در
  مدل~$\mModel M$ نشان می‌دهد~$R'[w][v]$. بنا بر فرض استقرا،
  $\mSat{M'}{!C}[[v]]$ و~$\mSat/{M'}{!D}[[v]]$؛ بنابراین
  $\mSat/{M'}{!C \lif !D}[[w]]$. حالت نفی همان تخصص این استدلال با
  $!D=\lfalse$ است.

  با اعمال لم تصفیه بر~$!A$ در~$w_0$، داریم
  $\mSat/{M'}{!A}[[w_0]]$. پس~$\mSat/{M'}{!A}$، و مدل متناهی مطلوب به
  دست آمد.
\end{proof}

\begin{prob}
جزئیات استقرا در اثبات~\olref[int][sc][dec]{thm:decidability} را برای
همهٔ !!{formula}های~$\Sigma$ کامل کنید و نشان دهید تعریف~$R'$ از انتخاب
نمایندگان مستقل است.
\end{prob}

از~\olref{thm:decidability} نتیجه می‌شود الگوریتمی برای تصمیم‌گیری دربارهٔ
$\Entails !A$ وجود دارد.
