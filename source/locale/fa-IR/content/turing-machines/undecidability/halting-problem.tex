\olsection{مسئلهٔ توقف}

\begin{explain}
فرض کنید برشمردنی از توصیف‌های ماشین‌های تورینگ تثبیت کرده‌ایم. بدین
ترتیب هر ماشین تورینگ یک \emph{نمایه} می‌گیرد: جایگاه آن در برشمردن
$M_1$، $M_2$، $M_3$،~\dots{} از توصیف‌های ماشین تورینگ.

می‌دانیم باید توابع تورینگ‌محاسبه‌ناپذیری وجود داشته باشند: مجموعهٔ
توصیف‌های ماشین‌های تورینگ، و بنابراین مجموعهٔ ماشین‌های تورینگ،
!!{enumerable} است، اما مجموعهٔ همهٔ توابع از~$\Nat$ به~$\Nat$
برشمردنی نیست. بااین‌حال می‌توانیم نمونه‌های مشخصی از توابع
محاسبه‌ناپذیر نیز بیابیم. یکی از آن‌ها تابع توقف است.
\end{explain}

\begin{defn}[تابع توقف]
 \emph{تابع توقف}~$h$ چنین تعریف می‌شود:
\[
h(e,n)=
\begin{cases}
  \text{0} & \text{اگر ماشین~$M_e$ برای ورودی~$n$ متوقف نشود} \\
  \text{1} & \text{اگر ماشین~$M_e$ برای ورودی~$n$ متوقف شود}
\end{cases}
\]
\end{defn}

\begin{defn}[مسئلهٔ توقف]
\emph{مسئلهٔ توقف} مسئلهٔ تعیین این است که (برای هر $e$ و~$n$) آیا
ماشین تورینگ~$M_e$ برای ورودیِ~$n$ خط متوقف می‌شود یا نه.
\end{defn}

\begin{explain}
با نشان‌دادن اینکه تابع مرتبطِ~$s$ تورینگ‌محاسبه‌پذیر نیست، نشان
می‌دهیم که~$h$ تورینگ‌محاسبه‌پذیر نیست. این اثبات بر این دو واقعیت
تکیه دارد که هرچه با ماشین تورینگ محاسبه شود با ماشین تورینگ منضبط
نیز محاسبه می‌شود (\olref[mac][dis]{sec}) و دو ماشین تورینگ را می‌توان
به هم متصل کرد تا ماشینی واحد ساخته شود (\olref[mac][cmb]{sec}).
\end{explain}

\begin{defn} تابع~$s$ چنین تعریف می‌شود:
\[
s(e)=
\begin{cases}
  \text{0} & \text{اگر ماشین~$M_e$ برای ورودی~$e$ متوقف نشود} \\
  \text{1} & \text{اگر ماشین~$M_e$ برای ورودی~$e$ متوقف شود}
\end{cases}
\]
\end{defn}

\begin{lem}
تابع~$s$ تورینگ‌محاسبه‌پذیر نیست.
\end{lem}

\begin{proof}
برای رسیدن به تناقض فرض کنید تابع~$s$ تورینگ‌محاسبه‌پذیر است. آنگاه
ماشین تورینگی چون~$S$ وجود می‌داشت که~$s$ را محاسبه کند. بدون کاستن
از کلیت می‌توانیم فرض کنیم $S$ هنگام توقف خانهٔ نخست را پویش می‌کند
(یعنی منضبط است). این ماشین را می‌توان به ماشین دیگری چون~$J$ «متصل
کرد». $J$ اگر با ورودی~$0$ آغاز شود متوقف می‌شود (یعنی در حالت آغازین
و هنگام پویش خانهٔ سمت راست نماد انتهای نوار، $\TMblank$ می‌خواند) و
در غیر این صورت به سمت راست دور می‌شود و هرگز متوقف نمی‌شود. ماشین
$S\concat J$ که از اتصال~$S$ به~$J$ ساخته شده است یک ماشین تورینگ است،
پس برای بعضی~$e$ همان~$M_e$ است (یعنی در جایی از برشمردن ظاهر می‌شود).
$M_e$ را با ورودیِ~$e$ نماد~$\TMstroke$ آغاز کنید. دو حالت ممکن است:
$M_e$ متوقف می‌شود یا متوقف نمی‌شود.
\begin{enumerate}
\item فرض کنید $M_e$ برای ورودیِ~$e$ نماد~$\TMstroke$ متوقف شود.
  آنگاه $s(e)=1$. پس $S$ پس از آغاز روی~$e$ با یک~$\TMstroke$ به‌عنوان
  خروجی روی نوار متوقف می‌شود. سپس $J$ با یک~$\TMstroke$ روی نوار آغاز
  می‌کند و در این حالت متوقف نمی‌شود. اما $M_e$ همان ماشین
  $S\concat J$ است، پس باید دقیقاً کاری را انجام دهد که اجرای~$S$ و
  سپس~$J$ انجام می‌دهد (یعنی در این حالت به سمت راست دور شود و هرگز
  متوقف نشود). پس $M_e$ نمی‌تواند برای ورودیِ~$e$ نماد~$\TMstroke$
  متوقف شود.

\item اکنون فرض کنید $M_e$ برای ورودیِ~$e$ نماد~$\TMstroke$ متوقف
  نشود. آنگاه $s(e)=0$ و~$S$ پس از آغاز با ورودی~$e$ با نواری خالی
  متوقف می‌شود. $J$ نیز پس از آغاز با نوار خالی بی‌درنگ متوقف می‌شود.
  باز هم $M_e$ همان کاری را می‌کند که~$S$ و سپس~$J$ می‌کنند، پس باید
  برای ورودیِ~$e$ نماد~$\TMstroke$ متوقف شود.
\end{enumerate}
در هر دو حالت با فرض خود به تناقض می‌رسیم. پس هیچ ماشین تورینگی چون
$S$ نمی‌تواند وجود داشته باشد: $s$ تورینگ‌محاسبه‌پذیر نیست.
\end{proof}

\begin{thm}[حل‌ناپذیری مسئلهٔ توقف]
\ollabel{thm:halting-problem} مسئلهٔ توقف حل‌ناپذیر است؛ یعنی تابع~$h$
تورینگ‌محاسبه‌پذیر نیست.
\end{thm}

\begin{proof}
فرض کنید $h$ تورینگ‌محاسبه‌پذیر باشد، مثلاً با ماشین تورینگ~$H$. در
این صورت می‌توانستیم با~$H$ ماشینی تورینگ بسازیم که~$s$ را محاسبه
کند: نخست رونوشتی از ورودی بسازید (که با نماد~$\TMblank$ جدا شده
باشد). سپس به آغاز بازگردید و~$H$ را اجرا کنید. روشن است که می‌توانیم
ماشینی برای کار نخست بسازیم (\cref{tur:mac:dis:prob:copier} را
ببینید)، و اگر $H$ وجود داشت می‌توانستیم آن را به چنین ماشین
رونویسی «متصل کنیم» و ماشین تازه‌ای به دست آوریم که تعیین کند آیا
$M_e$ روی ورودی~$e$ متوقف می‌شود؛ یعنی~$s$ را محاسبه کند. اما پیش‌تر
نشان داده‌ایم چنین ماشینی نمی‌تواند وجود داشته باشد. پس $h$ نیز
تورینگ‌محاسبه‌پذیر نیست.
\end{proof}

\begin{prob}
مسئلهٔ سه‌توقف (3-Halt) مسئلهٔ ارائهٔ روشی تصمیم‌گیر است که تعیین کند
آیا یک ماشین تورینگ دلخواه روی ورودیِ سه~$\TMstroke$ در نواری که در
بقیهٔ جاها خالی است متوقف می‌شود یا نه. ثابت کنید مسئلهٔ سه‌توقف
حل‌ناپذیر است.
\end{prob}

\begin{prob}
نشان دهید اگر مسئلهٔ توقف برای زوج‌های ماشین تورینگ و ورودیِ~$M_e$
و~$n$ که در آن‌ها $e\neq n$ حل‌پذیر باشد، برای حالت‌های $e=n$ نیز
حل‌پذیر است.
\end{prob}

\begin{prob}
ثابت کردیم مسئلهٔ توقف هنگامی حل‌ناپذیر است که ورودی عددی چون~$e$
باشد که از راه برشمردن همهٔ ماشین‌های تورینگ، ماشین~$M_e$ را مشخص
می‌کند. اگر اجازه دهیم توصیف‌های ماشین‌های تورینگ از
\olref[tur][und][enu]{sec} مستقیماً ورودی باشند چه؟ آیا ممکن است ماشین
تورینگی مسئلهٔ توقف را تصمیم کند اما به‌جای نمایه، توصیف ماشین‌های
تورینگ را ورودی بگیرد؟ توضیح دهید چرا یا چرا نه.
\end{prob}

\begin{prob} نشان دهید تابع \emph{جزئیِ}~$s'$ که به صورت زیر تعریف
  می‌شود
  \[
  s'(e)=
  \begin{cases}
    \text{1} & \text{اگر ماشین~$M_e$ برای ورودی~$e$ متوقف شود}\\
    \text{تعریف‌نشده} & \text{اگر ماشین~$M_e$ برای ورودی~$e$ متوقف نشود}
  \end{cases}
  \]
  تورینگ‌محاسبه‌پذیر \emph{است}.
\end{prob}
