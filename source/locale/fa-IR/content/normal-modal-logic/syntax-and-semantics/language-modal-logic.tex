\olsection{زبان منطق موجهات پایه}

\begin{defn}
زبان پایهٔ منطق موجهات شامل موارد زیر است:
\begin{enumerate}
  \tagitem{prvFalse}{ثابت گزاره‌ایِ !!{falsity}، یعنی~$\lfalse$.}{}
  \tagitem{prvTrue}{ثابت گزاره‌ایِ !!{truth}، یعنی~$\ltrue$.}{}
  \item مجموعه‌ای !!{denumerable} از !!{propositional variable}ها:
    $\Obj p_0$، $\Obj p_1$، $\Obj p_2$، \dots
  \item رابط‌های گزاره‌ای: \startycommalist
  \iftag{prvNot}{\ycomma $\lnot$ (نفی)}{}%
  \iftag{prvAnd}{\ycomma $\land$ (عطف)}{}%
  \iftag{prvOr}{\ycomma $\lor$ (فصل)}{}%
  \iftag{prvIf}{\ycomma $\lif$ (!!{conditional})}{}%
  \iftag{prvIff}{\ycomma $\liff$ (!!{biconditional})}{}.
  \tagitem{prvBox}{عملگر موجهاتی~$\Box$.}{}
  \tagitem{prvDiamond}{عملگر موجهاتی~$\Diamond$.}{}
\end{enumerate}
\end{defn}

\begin{defn}
\emph{!!^{formula}ها}ی زبان پایهٔ موجهات به‌طور استقرایی چنین تعریف
می‌شوند:
\begin{enumerate}
\tagitem{prvFalse}{$\lfalse$ یک !!{formula} اتمی است.}{}

\tagitem{prvTrue}{$\ltrue$ یک !!{formula} اتمی است.}{}

\item هر !!{propositional variable}~$\Obj p_i$ یک !!{formula} (اتمی) است.

\tagitem{prvNot}{اگر~$!A$ !!a{formula} باشد، آنگاه~$\lnot !A$
  نیز !!a{formula} است.}{}

\tagitem{prvAnd}{اگر~$!A$ و~$!B$ !!{formula} باشند، آنگاه
  $(!A \land !B)$ !!a{formula} است.}{}

\tagitem{prvOr}{اگر~$!A$ و~$!B$ !!{formula} باشند، آنگاه
  $(!A \lor !B)$ !!a{formula} است.}{}

\tagitem{prvIf}{اگر~$!A$ و~$!B$ !!{formula} باشند، آنگاه
  $(!A \lif !B)$ !!a{formula} است.}{}

\tagitem{prvIff}{اگر~$!A$ و~$!B$ !!{formula} باشند، آنگاه
  $(!A \liff !B)$ !!a{formula} است.}{}

\tagitem{prvBox}{اگر~$!A$ !!a{formula} باشد، آنگاه~$\Box !A$
  نیز !!a{formula} است.}{}

\tagitem{prvDiamond}{اگر~$!A$ !!a{formula} باشد، آنگاه~$\Diamond !A$
  نیز !!a{formula} است.}{}

\tagitem{limitClause}{هیچ عبارت دیگری !!a{formula} نیست.}{}
\end{enumerate}
\end{defn}

\begin{tagblock}{defNot,defOr,defAnd,defIf,defIff,defTrue,defFalse,defBox,defDiamond}
\begin{defn}
فرمول‌هایی که با عملگرهای تعریف‌شده ساخته می‌شوند باید به صورت زیر
فهمیده شوند:

\begin{tagenumerate}{defTrue,defFalse,defNot,defOr,defAnd,defIf,defIff,defBox,defDiamond}

\tagitem{defTrue}{$\ltrue$ مخففِ
  \iftag{prvFalse}{$\lnot\lfalse$}{$(!A \lor \lnot !A)$ برای یک
    !!{formula} اتمیِ ثابت~$!A$} است.}{}

\tagitem{defFalse}{$\lfalse$ مخففِ
  \iftag{prvTrue}{$\lnot\ltrue$}{$(!A \land \lnot !A)$ برای یک
    !!{formula} اتمیِ ثابت~$!A$} است.}{}

\tagitem{defNot}{$\lnot !A$ مخففِ~$!A \lif \lfalse$ است.}{}

\tagitem{defOr}{$!A \lor !B$ مخففِ
  \iftag{prvAnd}{$\lnot(\lnot !A \land \lnot !B)$}{$\lnot !A \lif
    !B$} است.}{}

\tagitem{defAnd}{$!A \land !B$ مخففِ
  \iftag{prvOr}{$\lnot(\lnot !A \lor \lnot !B)$}{$\lnot (!A \lif
    \lnot !B)$} است.}{}

\tagitem{defIf}{$!A \lif !B$ مخففِ
  \iftag{prvOr}{$(\lnot !A \lor !B)$}{$\lnot (!A \land \lnot !B)$} است.}{}

\tagitem{defIff}{$!A \liff !B$ مخففِ~$(!A \lif !B) \land (!B
  \lif !A)$ است.}{}

\tagitem{defBox}{$\Box !A$ مخففِ~$\lnot\Diamond\lnot !A$ است. }{}

\tagitem{defDiamond}{$\Diamond !A$ مخففِ~$\lnot\Box\lnot !A$ است.}{}
\end{tagenumerate}
\end{defn}
\end{tagblock}

اگر !!a{formula}~$!A$ شامل~$\Box$ یا~$\Diamond$ نباشد، آن را
\emph{فاقد وجه} می‌نامیم.
