\olsection{صدق در یک جهان}

هر مدل موجهاتی تعیین می‌کند کدام !!{formula} موجهاتی در کدام جهان‌های
آن صادق به شمار می‌آیند. رابطهٔ «مدل~$\mModel{M}$، !!{formula}~$!A$ را
در جهان~$w$ صادق می‌سازد» مفهوم پایهٔ معناشناسی رابطه‌ای است. این رابطه
به‌طور استقرایی تعریف می‌شود و برای عملگرهای غیرموجهاتی با توصیف معمول
به‌کمک جدول‌های ارزش منطبق است.

\begin{defn}\ollabel{defn:mmodels}
  \emph{صدق !!a{formula}~$!A$ در~$w$} در مدل~$\mModel M$، که با
  $\mSat{M}{!A}[w]$ نمادگذاری می‌شود، به‌طور استقرایی چنین تعریف می‌شود:
  \begin{enumerate}
  \tagitem{prvFalse}{\indcase{!A}{\lfalse}{$\mSat{M}{\lfalse}[w]$ هرگز برقرار نیست.}}{}
  \tagitem{prvTrue}{\indcase{!A}{\ltrue}{$\mSat{M}{\ltrue}[w]$ همواره برقرار است.}}{}
  \item $\mSat{M}{p}[w]$ اگر و تنها اگر~$w \in V(p)$.
  \tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$ اگر و تنها اگر
    $\mSat/{M}{!B}[w]$.}}{}
  \tagitem{prvAnd}{\indcase{!A}{(!B \land !C)}{$\mSat{M}{\indfrm}[w]$ اگر و تنها اگر
    $\mSat{M}{!B}[w]$ و~$\mSat{M}{!C}[w]$.}}{}
  \tagitem{prvOr}{\indcase{!A}{(!B \lor !C)}{$\mSat{M}{\indfrm}[w]$ اگر و تنها اگر
    $\mSat{M}{!B}[w]$ یا~$\mSat{M}{!C}[w]$} (یا هر دو).}{}
  \tagitem{prvIf}{\indcase{!A}{(!B \lif !C)}{$\mSat{M}{\indfrm}[w]$ اگر و تنها اگر
    $\mSat/{M}{!B}[w]$ یا~$\mSat{M}{!C}[w]$.}}{}
  \tagitem{prvIff}{\indcase{!A}{(!B \liff !C)}{$\mSat{M}{\indfrm}[w]$ اگر و تنها اگر
    یا هر دو رابطهٔ~$\mSat{M}{!B}[w]$ و~$\mSat{M}{!C}[w]$ برقرار باشند، یا
    هیچ‌یک از~$\mSat{M}{!B}[w]$ و~$\mSat{M}{!C}[w]$ برقرار نباشند.}}{}
  \item\ollabel{defn:sub:mmodels-box}
    \indcase{!A}{\Box !B}{$\mSat{M}{\indfrm}[w]$ اگر و تنها اگر
    برای هر~$w' \in W$ که~$Rww'$، رابطهٔ~$\mSat{M}{!B}[w']$ برقرار باشد.}
  \item\ollabel{defn:sub:mmodels-diamond}
    \indcase{!A}{\Diamond !B}{$\mSat{M}{\indfrm}[w]$ اگر و تنها اگر
    دست‌کم برای یک~$w' \in W$ که~$Rww'$، رابطهٔ~$\mSat{M}{!B}[w']$ برقرار باشد.}
  \end{enumerate}
\end{defn}

توجه کنید که بنا بر بند~\olref{defn:sub:mmodels-box}، هرگاه هیچ~$w'$ای
وجود نداشته باشد که~$Rww'$، !!a{formula}~$\Box !B$ در~$w$ صادق است.
در این حالت~$\Box !B$ در~$w$ \emph{به‌طور تهی} صادق است. همچنین ممکن
است~$\Box !B$ در~$w$ ارضا شود، بی‌آنکه~$!B$ در آنجا ارضا شود. صدق~$!B$
در~$w$ صدق~$\Diamond !B$ در~$w$ را تضمین نمی‌کند. بااین‌حال، اگر~$Rww$
برقرار باشد، مثلاً اگر~$R$ بازتابی باشد، این تضمین برقرار است. اگر
هیچ~$w'$ای وجود نداشته باشد که~$Rww'$، آنگاه برای هر~$!A$،
$\mSat/{M}{\Diamond !A}[w]$.

\begin{prob}
  مدلِ \olref[nml][syn][rel]{fig:simple} را در نظر بگیرید. کدام‌یک از
  موارد زیر برقرارند؟
    \begin{enumerate}
    \item $\mSat{M}{q}[w_1]$؛
    \item $\mSat{M}{\lnot q}[w_3]$؛
    \item $\mSat{M}{p \lor q}[w_1]$؛
    \item $\mSat{M}{\Box (p \lor q)}[w_1]$؛
    \item $\mSat{M}{\Box q}[w_3]$؛
    \item $\mSat{M}{\Box \bot}[w_3]$؛
    \item $\mSat{M}{\Diamond q}[w_1]$؛
    \item $\mSat{M}{\Box q}[w_1]$؛
    \item $\mSat{M}{\lnot \Box\Box \lnot q}[w_1]$.
    \end{enumerate}
\end{prob}

% Prove duality of \Box and \Diamond
\begin{tagblock}{prvDiamond}
\begin{prop}\ollabel{prop:dual}
  \begin{enumerate}
  \item $\mSat{M}{\Box !A}[w]$ اگر و تنها اگر~$\mSat{M}{\lnot\Diamond\lnot !A}[w]$.
  \item $\mSat{M}{\Diamond !A}[w]$ اگر و تنها اگر~$\mSat{M}{\lnot\Box\lnot !A}[w]$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ اگر و تنها اگر
      $\mSat/{M}{\Diamond\lnot !A}[w]$، بنا بر تعریف~$\mSat{M}{}[w]$.
      رابطهٔ~$\mSat{M}{\Diamond\lnot !A}[w]$ اگر و تنها اگر برای یک~$w'$
      با~$Rww'$، رابطهٔ~$\mSat{M}{\lnot !A}[w']$ برقرار باشد. پس
      $\mSat/{M}{\Diamond\lnot !A}[w]$ اگر و تنها اگر برای هر~$w'$ با
      $Rww'$، رابطهٔ~$\mSat/{M}{\lnot !A}[w']$ برقرار باشد. همچنین
      $\mSat/{M}{\lnot !A}[w']$ اگر و تنها اگر~$\mSat{M}{!A}[w']$.
      در نتیجه، $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ اگر و تنها اگر
      برای هر~$w'$ با~$Rww'$، رابطهٔ~$\mSat{M}{!A}[w']$ برقرار باشد.
      بار دیگر بنا بر تعریف~$\mSat{M}{}[w]$، این شرط اگر و تنها اگر
      $\mSat{M}{\Box !A}[w]$.
    \item \iftag{probDiamond}{تمرین.}{$\mSat{M}{\lnot\Box\lnot !A}[w]$
      اگر و تنها اگر~$\mSat/{M}{\Box\lnot !A}[w]$. رابطهٔ
      $\mSat{M}{\Box\lnot !A}[w]$ اگر و تنها اگر برای هر~$w'$ با
      $Rww'$، رابطهٔ~$\mSat{M}{\lnot !A}[w']$ برقرار باشد. پس
      $\mSat/{M}{\Box\lnot !A}[w]$ اگر و تنها اگر برای یک~$w'$ با
      $Rww'$، رابطهٔ~$\mSat/{M}{\lnot !A}[w']$ برقرار باشد. همچنین
      $\mSat/{M}{\lnot !A}[w']$ اگر و تنها اگر~$\mSat{M}{!A}[w']$.
      در نتیجه، $\mSat{M}{\lnot\Box\lnot !A}[w]$ اگر و تنها اگر برای
      یک~$w'$ با~$Rww'$، رابطهٔ~$\mSat{M}{!A}[w']$ برقرار باشد. بار دیگر
      بنا بر تعریف~$\mSat{M}{}[w]$، این شرط اگر و تنها اگر
      $\mSat{M}{\Diamond !A}[w]$.}
  \end{enumerate}
\end{proof}

\begin{probtag}{probDiamond}
  اثباتِ \olref[nml][syn][trw]{prop:dual} را کامل کنید.
\end{probtag}
\end{tagblock}

\begin{prob}
  فرض کنید~$\mModel{M} = \tuple{W, R, V}$ یک مدل باشد و
  $w_1, w_2 \in W$ چنان باشند که:
  \begin{enumerate}
  \item $w_1 \in V(p)$ اگر و تنها اگر~$w_2 \in V(p)$ (برای هر
    !!{propositional variable}~$p$)؛ و
  \item برای هر~$w \in W$، $Rw_1w$ اگر و تنها اگر~$Rw_2w$.
  \end{enumerate}
  با استفاده از استقرا بر !!{formula} نشان دهید که برای هر !!{formula}~$!A$،
  $\mSat{M}{!A}[w_1]$ اگر و تنها اگر~$\mSat{M}{!A}[w_2]$.
\end{prob}

\begin{probtag}{prvDiamond}
  فرض کنید~$\mModel{M} = \tuple{W, R, V}$. نشان دهید
  $\mSat{M}{\lnot\Diamond !A}[w]$ اگر و تنها اگر
  $\mSat{M}{\Box\lnot!A}[w]$.
\end{probtag}
