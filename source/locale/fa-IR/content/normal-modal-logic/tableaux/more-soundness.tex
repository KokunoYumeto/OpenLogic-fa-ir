\olsection{درستیِ قواعد افزوده}

می‌گوییم قاعده‌ای برای رده‌ای از مدل‌ها درست است اگر هرگاه شاخه‌ای در
!!a{tableau} در مدلی از آن رده ارضاپذیر باشد، شاخهٔ حاصل از اعمال قاعده
نیز در مدلی از آن رده ارضاپذیر باشد.

\begin{prop}
  \ollabel{prop:soundness-T} 
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ و }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{برای مدل‌های بازتابی درست است}{برای
  مدل‌های بازتابی درست‌اند}.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{شاخه با اعمال~T$\Box$ بر
  $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$ گسترش می‌یابد:
  \iftag{probBox}{تمرین.}{این کار !!{signed formula} تازهٔ
    $\sFmla{\True}{!B}[\sigma]$ را روی شاخه به دست می‌دهد. فرض کنید
    $\mSat{M}{\Gamma}[f]$، و به‌ویژه
    $\mSat{M}{\Box !B}[f(\sigma)]$. چون~$R$ بازتابی است، می‌دانیم
    $Rf(\sigma)f(\sigma)$. پس~$\mSat{M}{!B}[f(\sigma)]$؛ یعنی
    $\mModel{M}, f$، $\sFmla{\True}{!B}[\sigma]$ را ارضا می‌کند.}}{}
\tagitem{prvDiamond}{شاخه با اعمال~T$\Diamond$ بر
  $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$ گسترش می‌یابد:
  \iftag{probDiamond}{تمرین.}{این کار !!{signed formula} تازهٔ
    $\sFmla{\False}{!B}[\sigma]$ را روی شاخه به دست می‌دهد. فرض کنید
    $\mSat{M}{\Gamma}[f]$، و به‌ویژه
    $\mSat/{M}{\Diamond !B}[f(\sigma)]$. چون~$R$ بازتابی است، می‌دانیم
    $Rf(\sigma)f(\sigma)$. پس~$\mSat/{M}{!B}[f(\sigma)]$؛ یعنی
    $\mModel{M}, f$، $\sFmla{\False}{!B}[\sigma]$ را ارضا می‌کند.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  اثبات \olref[nml][tab][msn]{prop:soundness-T} را کامل کنید.
\end{prob}
\tagendprob
