\olsection{عملگرهای موجهاتی و مجموعه‌های کامل سازگار}

\begin{explain}
هنگامی که مدلی~$\mModel{M^\Sigma}$ می‌سازیم که مجموعهٔ جهان‌هایش از
مجموعه‌های کاملِ~$\Sigma$-سازگار~$\Delta$ در یک منطق موجهات عادی
$\Sigma$ تشکیل می‌شود، باید رابطهٔ دسترسی‌پذیری~$R^\Sigma$ میان چنین
«جهان‌هایی» را نیز تعریف کنیم. می‌خواهیم رابطهٔ دسترسی‌پذیری (و
تخصیص~$V^\Sigma$) چنان تعریف شوند که
$\mSat{M^\Sigma}{!A}[\Delta]$ اگر و تنها اگر~$!A \in \Delta$. چگونه
باید چنین کنیم؟

پس از تعریف رابطهٔ دسترسی‌پذیری، تعریف صدق در یک جهان تضمین می‌کند که
\iftag{prvBox}{$\mSat{M^\Sigma}{\Box!A}[\Delta]$ اگر و تنها اگر برای
  همهٔ~$\Delta'$هایی که~$R^\Sigma\Delta\Delta'$،
  $\mSat{M^\Sigma}{!A}[\Delta']$}{$\mSat{M^\Sigma}{\Diamond!A}[\Delta]$
  اگر و تنها اگر دست‌کم برای یک~$\Delta'$ که~$R^\Sigma\Delta\Delta'$،
  $\mSat{M^\Sigma}{!A}[\Delta']$}. اثبات اینکه
$\mSat{M^\Sigma}{!A}[\Delta]$ اگر و تنها اگر~$!A \in \Delta$، مستلزم

\begin{defn}
  اگر~$\Gamma$ مجموعه‌ای از !!{formula}هاست، بگذارید
  \begin{align*}
    \Box\Gamma & = \Setabs{\Box !B}{!B \in \Gamma}\\
    \Diamond\Gamma & = \Setabs{\Diamond !B}{!B \in \Gamma}\\
    \intertext{و}
    \Box^{-1}\Gamma & = \Setabs{!B}{\Box !B \in \Gamma}\\
    \Diamond^{-1}\Gamma & = \Setabs{!B}{\Diamond !B \in \Gamma}\\
  \end{align*}
\end{defn}

به بیان دیگر، $\Box\Gamma$ همان~$\Gamma$ است با~$\Box$ در ابتدای هر
!!{formula} آن؛ و~$\Box^{-1}\Gamma$ از همهٔ !!{formula}های دارای
$\Box$ در~$\Gamma$ تشکیل می‌شود که~$\Box$ آغازینشان حذف شده است. این
تعریف به‌خودیِ‌خود اهمیت چندانی ندارد، اما نمادگذاری را به‌طور چشمگیری
ساده می‌کند.

توجه کنید که~$\Box\Box^{-1}\Gamma \subseteq \Gamma$:
\[
\Box\Box^{-1}\Gamma = \Setabs{\Box!B}{\Box!B \in \Gamma}
\]
یعنی صرفاً مجموعهٔ همهٔ آن !!{formula}های~$\Gamma$ است که با~$\Box$
آغاز می‌شوند.

\begin{lem}\ollabel{lem:box1}
  اگر~$\Gamma \Proves[\Sigma] !A$، آنگاه~$\Box\Gamma
  \Proves[\Sigma] \Box !A$.
\end{lem}

\begin{proof}
  اگر~$\Gamma \Proves[\Sigma] !A$، آنگاه~$!B_1$، \dots، $!B_k
  \in \Gamma$ وجود دارند چنان‌که~$\Sigma \Proves !B_1 \lif
  (!B_2 \lif \cdots (!B_k \lif !A)\cdots)$. چون~$\Sigma$ عادی است،
  بنا بر قاعدهٔ~\RK{}،
  $\Sigma \Proves \Box!B_1 \lif (\Box!B_2 \lif \cdots
  (\Box!B_k \lif \Box!A)\cdots)$، و آشکارا~$\Box!B_1$، \dots،
  $\Box!B_k \in \Box\Gamma$. پس بنا به تعریف،
  $\Box\Gamma \Proves[\Sigma] \Box !A$.
\end{proof}

\begin{lem}\ollabel{lem:box2}
  اگر~$\Box^{-1} \Gamma \Proves[\Sigma] !A$، آنگاه~$\Gamma
  \Proves[\Sigma] \Box!A$.
\end{lem}

\begin{proof}
  فرض کنید~$\Box^{-1}\Gamma \Proves[\Sigma] !A$؛ آنگاه بنا بر
  \olref{lem:box1}، $\Box\Box^{-1}\Gamma \Proves[\Sigma] \Box!A$.
  اما چون~$\Box\Box^{-1}\Gamma \subseteq \Gamma$، بنا بر یکنوایی
  همچنین~$\Gamma \Proves[\Sigma] \Box!A$.
\end{proof}

  $\Delta$ که~$\Diamond\Delta \subseteq \Gamma$، داشته باشیم
  $!A \in \Delta$.

  $\Box\lnot !A \notin \Gamma$. چون~$\Gamma$ کاملِ~$\Sigma$-سازگار

\begin{lem}\ollabel{lem:box-iff-diamond}
  فرض کنید~$\Gamma$ و~$\Delta$ کاملِ~$\Sigma$-سازگار باشند. آنگاه
  $\Box^{-1}\Gamma \subseteq \Delta$ اگر و تنها اگر
  $\Diamond\Delta \subseteq \Gamma$.
\end{lem}

\begin{proof}
  جهت «تنها اگر»: فرض کنید~$\Box^{-1}\Gamma \subseteq \Delta$ و
  $\Diamond!A \in \Diamond\Delta$ (یعنی~$!A \in \Delta$). برای
  نشان‌دادن~$\Diamond!A \in \Gamma$ کافی است نشان دهیم
  $\Box\lnot!A \notin \Gamma$، زیرا آنگاه بنا بر بیشینگی
  $\lnot\Box\lnot !A \in \Gamma$. حال اگر~$\Box\lnot!A \in
  \Gamma$، آنگاه بنا بر فرض~$\lnot!A \in \Delta$، که با سازگاری
  $\Delta$ ناسازگار است (زیرا~$!A \in \Delta$). پس چنان‌که لازم بود
  $\Box\lnot!A \notin \Gamma$.

  جهت «اگر»: فرض کنید~$\Diamond\Delta \subseteq \Gamma$. به روش عکس
  نقیض استدلال می‌کنیم: فرض کنید~$!A \notin \Delta$ تا نشان دهیم
  $\Box!A \notin \Gamma$. اگر~$!A \notin \Delta$، آنگاه بنا بر
  بیشینگی~$\lnot!A \in \Delta$ و بنابراین بنا بر فرض
  $\Diamond\lnot!A \in \Gamma$. اما در هر منطق موجهات عادی،
  $\Diamond\lnot!A$ با~$\lnot\Box !A$ هم‌ارز است؛ و اگر دومی عضو
  $\Gamma$ باشد، بنا بر سازگاری~$\Box!A \notin\Gamma$، چنان‌که لازم
  بود.
\end{proof}
