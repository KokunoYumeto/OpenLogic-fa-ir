\olsection{نمونه‌هایی از تصفیه‌ها}

هنوز نشان نداده‌ایم که اصلاً تصفیه‌ای وجود دارد. اما در واقع، برای هر
مدل~$\mModel{M}$، تصفیه‌های بسیاری از~$\mModel{M}$ از رهگذر~$\Gamma$
وجود دارند. به‌طور خاص، دو تای آنها را مشخص می‌کنیم: ریزترین و
درشت‌ترین تصفیه. تصفیه‌های یک مدل واحد در رابطهٔ دسترسی خود تفاوت
خواهند داشت (زیرا \olref[fil]{defn:filtration} مستقیماً مقرر می‌کند که
$W^*$ و~$V^*$ چه باشند). ریزترین تصفیه تا حد امکان کمترین جهان‌های
مرتبط را خواهد داشت، حال آنکه درشت‌ترین تصفیه تا حد امکان بیشترین را.

\begin{defn}
  هرگاه $\Gamma$ تحت زیرفرمول‌ها بسته باشد، \emph{ریزترین}
  تصفیهٔ~$\mModel{M^*}$ یک مدل~$\mModel{M}$ با قرار دادن رابطهٔ زیر
  تعریف می‌شود:
  \[
  R^*[u][v] \quad \text{اگر و تنها اگر} \quad \exists u'\in [u] \;
  \exists v' \in [v] : Ru'v'.
  \]
\end{defn}

\begin{prop}\ollabel{prop:finest}
  ریزترین تصفیهٔ~$\mModel{M^*}$ به‌راستی یک تصفیه است.
\end{prop}

\begin{proof}
  باید بررسی کنیم که $R^*$، با این تعریف، شرط
  \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R} را برآورده
  می‌کند. سه شرط را به‌ترتیب بررسی می‌کنیم.

  اگر $Ruv$، چون $u \in [u]$ و $v \in [v]$، همچنین $R^*[u][v]$؛ پس
  \olref[fil]{defn:filtration-R1} برقرار است.
  
  \iftag{prvBox}{\iftag{probBox}{بررسی
      \olref[fil]{defn:filtration-R2} را به‌عنوان تمرین وا می‌گذاریم.}{برای
        \olref[fil]{defn:filtration-R2}، فرض کنید $\Box!A \in \Gamma$،
        $R^*[u][v]$ و $\mSat{M}{\Box!A}[u]$. بنا بر تعریف~$R^*$،
        $u' \equiv u$ و $v' \equiv v$هایی وجود دارند به‌گونه‌ای که
        $Ru'v'$. چون $u$ و $u'$ دربارهٔ~$\Gamma$ توافق دارند،
        $\mSat{M}{\Box!A}[u']$ نیز برقرار است؛ بنابراین
        $\mSat{M}{!A}[v']$. از بسته‌بودن~$\Gamma$ تحت
        زیر-!!{formula}ها، $v$ و $v'$ دربارهٔ~$!A$ توافق دارند؛ پس،
        چنان‌که مطلوب بود، $\mSat{M}{!A}[v]$.}}{}
  
  \iftag{prvDiamond}{\iftag{probDiamond}{بررسی
      \olref[fil]{defn:filtration-R3} را به‌عنوان تمرین وا می‌گذاریم.}{برای بررسی
        \olref[fil]{defn:filtration-R3}، فرض کنید $\Diamond!A \in
        \Gamma$، $R^*[u][v]$ و $\mSat{M}{!A}[v]$. بنا بر تعریف~$R^*$،
        $u' \equiv u$ و $v' \equiv v$هایی وجود دارند به‌گونه‌ای که
        $Ru'v'$. چون $v$ و $v'$ دربارهٔ~$\Gamma$ توافق دارند و
        $\Gamma$ تحت زیر-!!{formula}ها بسته است،
        $\mSat{M}{!A}[v']$ نیز برقرار است؛ پس
        $\mSat{M}{\Diamond !A}[u']$. چون $u$ و $u'$ نیز دربارهٔ
        $\Gamma$ توافق دارند، $\mSat{M}{\Diamond !A}[u]$.}}{}
\end{proof}

\begin{probtag}{probBox,probDiamond}
  اثبات \olref[nml][fil][exf]{prop:finest} را کامل کنید.
\end{probtag}

\begin{defn}
  هرگاه $\Gamma$ تحت زیرفرمول‌ها بسته باشد، \emph{درشت‌ترین}
  تصفیهٔ~$\mModel{M^*}$ یک مدل~$\mModel{M}$ با قرار دادن $R^*[u][v]$
  تعریف می‌شود، اگر و تنها اگر
  \iftag{notprvBox,notprvDiamond}{شرط زیر
    برقرار باشد:}{\emph{هر دو} شرط زیر برقرار باشند:}
  \begin{tagenumerate}{prvBox,prvDiamond}
  \tagitem{prvBox}{\ollabel{defn:coarsest-Box}اگر $\Box!A \in \Gamma$
    و $\mSat{M}{\Box!A}[u]$، آنگاه
    $\mSat{M}{!A}[v]$\iftag{prvDiamond}{؛}{.}}{}
  \tagitem{prvDiamond}{\ollabel{defn:coarsest-Diamond}اگر $\Diamond!A
    \in \Gamma$ و $\mSat{M}{!A}[v]$، آنگاه
    $\mSat{M}{\Diamond!A}[u]$.}{}
  \end{tagenumerate}
\end{defn}

\begin{prop}
  درشت‌ترین تصفیهٔ~$\mModel{M^*}$ به‌راستی یک تصفیه است.
\end{prop}

\begin{proof}
  با توجه به تعریف~$R^*$، تنها شرطی که بررسی آن باقی می‌ماند استلزام
  $R^*[u][v]$ از $Ruv$ است. پس $Ruv$ را فرض کنید.
  \iftag{prvBox}{فرض کنید $\Box!A \in \Gamma$ و
    $\mSat{M}{\Box!A}[u]$؛ آنگاه آشکارا
    $\mSat{M}{!A}[v]$\iftag{prvDiamond}{، و
      \olref{defn:coarsest-Box} برقرار
      است. }{.}}{}\iftag{prvDiamond}{فرض کنید $\Diamond!A \in
    \Gamma$ و $\mSat{M}{!A}[v]$. آنگاه $\mSat{M}{\Diamond!A}[u]$،
    زیرا~$Ruv$\iftag{prvBox}{، و \olref{defn:coarsest-Diamond}
      برقرار است}{}.}{}
\end{proof}

\begin{ex}
  فرض کنید $W = \PosInt$، $Rnm$ اگر و تنها اگر $m = n + 1$، و
  $V(p) = \Setabs{2n}{n \in \PosInt}$. مدل
  $\mModel{M} = \tuple{W, R, V}$ در \olref{fig:ex-filtration} ترسیم
  شده است. جهان‌ها $1$، $2$ و به همین ترتیب‌اند؛ هر جهان دقیقاً به یک
  جهان دیگر---جانشینش---دسترسی دارد، و $p$ در همه و تنها اعداد زوج
  صادق است.

  \begin{figure}
    \centering
    \begin{tikzpicture}[modal]
      \node[world] (1) [label=below:\mFalse{p}] {$1$};
      \node[world] (2) [label=below:\mTrue{p}, right=of 1] {$2$};
      \node[world] (3) [label=below:\mFalse{p}, right=of 2] {$3$};
      \node[world] (4) [label=below:\mTrue{p}, right=of 3] {$4$};
      \node[phantom] (5) [right=of 4] {};
      \draw[->] (1) to (2);
      \draw[->] (2) to (3);
      \draw[->] (3) to (4);
      \draw[dotted] (4) to (5);
    \end{tikzpicture}

    \begin{tikzpicture}[modal]
      \node[world] (1) [label=below:\mFalse{p}] {$[1]$};
      \node[world] (2) [label=below:\mTrue{p}, right=of 1] {$[2]$};
      \draw[->,bend left] (1) to (2);
      \draw[->,bend left] (2) to (1);
      
      \node[world] (3) [label=below:\mFalse{p},right=of 2] {$[1]$};
      \node[world] (4) [label=below:\mTrue{p}, right=of 3] {$[2]$};
      \draw[->,bend left] (3) to (4);
      \draw[->,bend left] (4) to (3);
      \draw[->,reflexive above] (4) to (4);
    \end{tikzpicture}
    \caption{یک مدل نامتناهی و تصفیه‌های آن.}
    \ollabel{fig:ex-filtration}
  \end{figure}
  
  اکنون فرض کنید $\Gamma$ مجموعهٔ زیر-!!{formula}های~$\Box p \lif p$،
  یعنی $\{p, \Box p, \Box p \lif p\}$ باشد. $p$ در همه و تنها اعداد
  زوج صادق است، و $\Box p$ در همه و تنها اعداد فرد؛ بنابراین
  $\Box p \lif p$ در همه و تنها اعداد زوج صادق است. به بیان دیگر، هر
  عدد فرد $\Box p$ را صادق، اما $p$ و $\Box p \lif p$ را کاذب می‌کند؛
  هر عدد زوج $p$ و $\Box p \lif p$ را صادق، اما $\Box p$ را کاذب
  می‌کند. پس $W^* = \{ [1], [2] \}$، که در آن $[1] =
  \{1, 3, 5, \dots\}$ و $[2] = \{2, 4, 6, \dots\}$. چون $2 \in
  V(p)$، $[2] \in V^*(p)$؛ و چون $1 \notin V(p)$، $[1] \notin
  V^*(p)$. پس $V^*(p) = \{[2]\}$.

  رابطهٔ دسترسی هر تصفیهٔ مبتنی بر~$W^*$ باید
  $\tuple{[1], [2]}, \tuple{[2],[1]}$ را دربرگیرد: چون $R12$، بنا بر
  \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R1} باید
  $R^*[1][2]$؛ و چون $R23$، باید $R^*[2][3]$، و $[3]=[1]$. این رابطه
  نمی‌تواند $\tuple{[1],[1]}$ را دربرگیرد: اگر چنین می‌کرد،
  $R^*[1][1]$، $\mSat{M}{\Box p}[1]$ اما $\mSat/{M}{p}[1]$ می‌داشتیم
  که با \olref[fil]{defn:filtration-R2} ناسازگار است. هیچ چیز
  $R^*[2][2]$ را ایجاب یا منع نمی‌کند. پس دو تصفیهٔ ممکن از
  ~ $\mModel{M}$ وجود دارد که متناظرند با دو رابطهٔ دسترسی
  \[
  \{\tuple{[1],[2]}, \tuple{[2],[1]}\} \text{ و }
  \{\tuple{[1],[2]}, \tuple{[2],[1]}, \tuple{[2],[2]}\}.
  \]
  در هر دو حالت، $p$ و $\Box p \lif p$ در~$[1]$ کاذب و $\Box p$
  صادق‌اند؛ $p$ و $\Box p \lif p$ در~$[2]$ صادق و $\Box p$ کاذب است.
\end{ex}


\begin{prob}
  مدل زیر $\mModel{M} = \tuple{W, R, V}$ را در نظر بگیرید که در آن
  $W = \Setabs{0\sigma}{\sigma \in \Bin^*}$، مجموعهٔ دنباله‌های صفر و
  یک است که با صفر آغاز می‌شوند؛ $R\sigma\sigma'$ اگر و تنها اگر
  $\sigma' = \sigma 0$ یا $\sigma' = \sigma 1$؛ و
  $V(p) = W \cap \Setabs{\sigma 0}{\sigma \in \Bin^*}$ و
  $V(q) = W \cap \Setabs{\sigma 1}{\sigma \in
    \Bin^* \setminus \{1\}}$. تصویر آن چنین است:

  برقرار است.

  فرض کنید $\Gamma$ مجموعهٔ زیر-!!{formula}های~$\Box(p \lor q) \lif
  (\Box p \lor \Box q)$ باشد. $W^*$ و~$V^*$ چه هستند؟ رابطهٔ دسترسی
  ریزترین تصفیهٔ~$\mModel{M}$ چیست؟ رابطهٔ دسترسی درشت‌ترین تصفیه چطور؟
\end{prob}
