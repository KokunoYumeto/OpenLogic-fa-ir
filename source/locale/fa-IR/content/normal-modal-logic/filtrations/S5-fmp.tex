\olsection{\Log{K} و~\Log{S5} خاصیت مدل متناهی دارند}

\begin{defn}
  گفته می‌شود دستگاه منطق موجهاتی~$\Sigma$ دارای \emph{خاصیت مدل
    متناهی} است اگر هرگاه !!a{formula}~$!A$ در جهانی از مدلی برای
  $\Sigma$ صادق باشد، آنگاه~$!A$ در جهانی از یک مدل \emph{متناهی}
  برای~$\Sigma$ نیز صادق باشد.
\end{defn}

\begin{prop}\ollabel{prop:K-fmp}
  \Log{K} خاصیت مدل متناهی دارد.
\end{prop}

\begin{proof}
  \Log{K} مجموعهٔ !!{formula}های معتبر است؛ یعنی هر مدلی یک مدل برای
  \Log{K} است. بنا بر \olref[fil]{thm:filtrations}، اگر
  $\mSat{M}{!A}[w]$، آنگاه~$\mSat{M^*}{!A}[{[w]}]$ برای هر تصفیهٔ
  $\mModel{M}$ از خلال مجموعهٔ~$\Gamma$ از زیر-!!{formula}های~$!A$.
  هر !!{formula} تنها شمار متناهی‌ای زیر-!!{formula} دارد، پس~$\Gamma$
  متناهی است. بنا بر \olref[fin]{prop:filt-are-finite}،
  $\card{W^*} \le 2^n$، که در آن~$n$ شمار !!{formula}های~$\Gamma$
  است. و چون~\Log{K} هیچ محدودیتی بر مدل‌ها تحمیل نمی‌کند،
  $\mModel{M^*}$ یک مدل برای~\Log{K} است.
\end{proof}

برای نشان‌دادن اینکه منطق~\Log{L} با استفاده از تصفیه‌ها خاصیت مدل
متناهی دارد، ضروری است که تصفیهٔ هر مدل برای~\Log{L} خود نیز مدلی برای
\Log{L} باشد. این امر اغلب به کار قابل‌توجهی نیاز دارد و هر تصفیه‌ای
مدلی برای~\Log{L} به دست نمی‌دهد. بااین‌حال، این نتیجه برای مدل‌های
همگانی همچنان برقرار است.

\begin{prop}\ollabel{prop:univ-fin}
  بگذارید~$\mClass{U}$ ردهٔ مدل‌های همگانی باشد (بنگرید به
  \olref[frd][es5]{prop:S5=univ}) و~$\mClass{U}_\mathrm{Fin}$ ردهٔ همهٔ
  مدل‌های همگانی متناهی. آنگاه هر !!{formula}~$!A$ در~$\mClass{U}$
  معتبر است اگر و تنها اگر در~$\mClass{U}_\mathrm{Fin}$ معتبر باشد.
\end{prop}

\begin{proof}
  مدل‌های همگانی متناهی، مدل‌های همگانی‌اند؛ پس جهت چپ‌به‌راست بدیهی
  است. برای جهت راست‌به‌چپ، فرض کنید~$!A$ در جهان~$w$ از یک مدل
  همگانی~$\mModel{M}$ کاذب باشد. بگذارید~$\Gamma$ شامل~$!A$ و همهٔ
  زیرفرمول‌های آن باشد؛ آشکارا~$\Gamma$ متناهی است. تصفیهٔ
  $\mModel{M^*}$ از~$\mModel{M}$ را در نظر بگیرید؛ آنگاه بنا بر
  \olref[fin]{prop:filt-are-finite}، $\mModel{M^*}$ متناهی است، و بنا
  بر \olref[fil]{thm:filtrations}، $!A$ در~$[w]$ از~$\mModel{M^*}$
  کاذب است. تنها باید مشاهده کنیم که~$\mModel{M^*}$ نیز همگانی است:
  برای~$u$ و~$v$ مفروض، بنا بر فرض~$Ruv$، و بنا بر
  \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R}، همچنین
  $R^*[u][v]$.
\end{proof}

\begin{cor}\ollabel{cor:S5fmp}
  \Log{S5} خاصیت مدل متناهی دارد.
\end{cor}

\begin{proof}
  بنا بر \olref[frd][es5]{prop:S5=univ}، اگر~$!A$ در جهانی از یک مدل
  بازتابی و اقلیدسی صادق باشد، آنگاه در جهانی از یک مدل همگانی صادق
  است. بنا بر \olref{prop:univ-fin}، در جهانی از یک مدل همگانی متناهی
  صادق است (یعنی تصفیهٔ مدل از خلال مجموعهٔ زیر-!!{formula}های~$!A$).
  هر مدل همگانی همچنین بازتابی و اقلیدسی است؛ پس~$!A$ در جهانی از یک
  مدل متناهیِ بازتابی و اقلیدسی صادق است.
\end{proof}

\begin{prob}
  نشان دهید هر تصفیهٔ یک مدل سریال یا بازتابی، به‌ترتیب سریال یا
  بازتابی نیز هست.
\end{prob}

\begin{prob}
  تصفیه‌ای نامتقارن (نامتعدی، نااقلیدسی) از یک مدل متقارن (متعدی،
  اقلیدسی) بیابید.
\end{prob}
