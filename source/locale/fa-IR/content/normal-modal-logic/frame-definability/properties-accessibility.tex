\olsection{خواص روابط دسترسی}

بسیاری از !!{formula}های موجهاتی شاخص خواصی ساده و حتی آشنا از رابطهٔ
دسترسی‌اند. در یک جهت، این بدان معناست که هر مدلی با خاصیتی معین،
!!{formula} متناظر (و همهٔ نمونه‌های جانشینی آن) را صادق می‌سازد. با
پنج نمونهٔ کلاسیک از گونه‌های روابط دسترسی و فرمول‌هایی آغاز می‌کنیم
که صدقشان به‌وسیلهٔ آن روابط تضمین می‌شود.

\begin{thm}\ollabel{thm:soundschemas}
  فرض کنید~$\mModel{M} = \tuple{W, R, V}$ یک مدل باشد. اگر~$R$ خاصیت
  سمت چپِ \olref{tab:five} را داشته باشد، هر نمونهٔ !!{formula} سمت
  راست در~$\mModel{M}$ صادق است.
\end{thm}

\begin{table}[t]
    \begin{tabular}{| l || l |}
      \hline
      {\emph{اگر~$R$ چنین باشد:}} & {\emph{آنگاه این عبارت در~$\mModel{M}$ صادق است:}} \\
      \hline \hline
      \emph{مسلسل}: $\forall u \exists v Ruv$ & \hbox to.43\textwidth
           {$\Box p \lif \Diamond p$ \hfill (\Ax{D})} \\
      \hline
      \emph{بازتابی}: $\forall w Rww$  
      & \hbox to.43\textwidth
          {$\Box p \lif p$ \hfill (\Ax{T})} \\
      \hline
      \emph{متقارن}: &  \hbox to .43\textwidth
           {$p \lif \Box\Diamond p$ \hfill (\Ax{B})} \\
           $\forall u\forall v(Ruv \lif Rvu)$ & \\
      \hline
      \emph{متعدی}: & \hbox to.43\textwidth
           {$\Box p \lif \Box \Box p$ \hfill (\Ax{4})} \\
      $\forall u \forall v \forall w ((Ruv \land Rvw) \lif Ruw)$ & \\
      \hline 
      \emph{اقلیدسی}: & \hbox to .43\textwidth
        {$\Diamond p \lif \Box\Diamond p$ \hfill (\Ax{5})} \\
      $\forall w \forall u \forall v ((Rwu \land Rwv) \lif Ruv)$ &\\
      \hline
    \end{tabular}
    \caption{پنج واقعیت تناظر.}
    \ollabel{tab:five}
  \end{table} 


\begin{proof}
  حالت~\Ax{B} چنین است: برای نشان‌دادن اینکه طرحواره در یک مدل صادق
  است، باید نشان دهیم همهٔ نمونه‌هایش در همهٔ جهان‌های مدل صادق‌اند.
  پس~$!A \lif \Box\Diamond !A$ را نمونه‌ای معین از~\Ax{B} و
  $w \in W$ را جهانی دلخواه بگیرید. برای نشان‌دادن اینکه
  $\Box \Diamond !A$ در~$w$ صادق است، فرض کنید مقدم~$!A$ در~$w$ صادق
  باشد. پس باید نشان دهیم~$\Diamond !A$ در همهٔ~$w'$های دسترس‌پذیر
  از~$w$ صادق است. حال برای هر~$w'$ چنان‌که~$Rww'$، با استفاده از فرض
  تقارن، $Rw'w$ نیز برقرار است (نگاه کنید به~\olref{fig:Bsymm}). چون
  $\mSat{M}{!A}[w]$، داریم~$\mSat{M}{\Diamond !A}[w']$. چون~$w'$ یک
  جهان دلخواه با~$Rww'$ بود، $\mSat{M}{\Box\Diamond!A}[w]$.

  حالت‌های دیگر را به‌صورت تمرین واگذار می‌کنیم.
\end{proof}

\begin{prob}
  اثباتِ \olref[nml][frd][acc]{thm:soundschemas} را کامل کنید.
\end{prob}

توجه کنید که عکس استلزام‌های~\olref{thm:soundschemas} برقرار نیست: درست
نیست که اگر مدلی یک طرحواره را تصدیق کند، رابطهٔ دسترسی آن مدل خاصیت
متناظر را داشته باشد. در مورد~\Ax{T} و مدل‌های بازتابی، به‌آسانی می‌توان
نمونه‌ای از مدلی داد که خود~\Ax{T} در آن شکست بخورد: بگذارید
$W = \{w\}$، $R = \emptyset$ و~$V(p) = \emptyset$. در این صورت~$R$
بازتابی نیست، اما~$\mSat{M}{\Box p}[w]$ و~$\mSat/{M}{p}[w]$. ولی در
اینجا فقط یک نمونهٔ~\Ax{T} در~$\mModel{M}$ شکست می‌خورد؛ نمونه‌های دیگر،
مانند~$\Box \lnot p \lif \lnot p$، صادق‌اند. یافتن مثال‌هایی که در
آن‌ها \emph{هر نمونهٔ جانشینی}~\Ax{T} در~$\mModel{M}$ صادق باشد و
$\mModel{M}$ بازتابی نباشد دشوارتر است. بااین‌حال چنین مدل‌هایی نیز
وجود دارند:

\begin{prop}\ollabel{prop:reflexive}
  فرض کنید~$\mModel{M} = \tuple{W, R, V}$ مدلی باشد چنان‌که
  $W = \{u, v\}$، $u \neq v$ و
  $R = \{\tuple{u,v},\tuple{v,u}\}$. فرض کنید برای هر~$p$:
  $u \in V(p) \Leftrightarrow v \in V(p)$. آنگاه:
  \begin{enumerate}
  \item برای هر~$!A$، $\mSat{M}{!A}[u]$ اگر و تنها اگر
    $\mSat{M}{!A}[v]$ (از استقرا بر~$!A$ استفاده کنید).
  \item هر نمونهٔ~\Ax{T} در~$\mModel{M}$ صادق است.
  \end{enumerate}
  چون~$\mModel{M}$ بازتابی نیست (در واقع \emph{نابازتابی} است)، عکسِ
  \olref{thm:soundschemas} در مورد~\Ax{T} شکست می‌خورد (برای برخی، هرچند
  نه همهٔ، طرحواره‌های دیگرِ مذکور در~\olref{thm:soundschemas} نیز
  می‌توان استدلال‌هایی مشابه ارائه کرد).
\end{prop}

\begin{prob}
  ادعاهای~\olref[nml][frd][acc]{prop:reflexive} را اثبات کنید.
\end{prob}

گرچه بر پنج !!{formula} کلاسیک~\Ax{D}، \Ax{T}، \Ax{B}، \Ax{4} و
\Ax{5} تمرکز خواهیم کرد، در~\olref{tab:anotherfive} چند خاصیت دیگر از
روابط دسترسی را ثبت می‌کنیم. رابطهٔ دسترسی~$R$ نیمه‌تابعی است اگر از هر
جهان حداکثر یک جهان دسترس‌پذیر باشد. اگر از هر جهان دقیقاً یک جهان
دسترس‌پذیر باشد، آن را تابعی می‌نامیم. (پس روابط تابعی دقیقاً روابطی‌اند
که هم مسلسل و هم نیمه‌تابعی‌اند.) آن‌ها را «تابعی» می‌نامند زیرا رابطهٔ
دسترسی مانند یک تابع (جزئی) عمل می‌کند. رابطه‌ای ضعیفاً چگال است اگر
هرگاه~$Ruv$، جهانی~$w$ «میان»~$u$ و~$v$ وجود داشته باشد. ازاین‌رو روابط
ضعیفاً چگال از جهتی عکس روابط متعدی‌اند: در رابطه‌ای متعدی، هرگاه بتوان
از~$u$ با مسیر غیرمستقیم از راه~$w$ به~$v$ رسید، می‌توان مستقیماً نیز
از~$u$ به~$v$ رسید؛ در رابطه‌ای ضعیفاً چگال، هرگاه بتوان مستقیماً از
$u$ به~$v$ رسید، می‌توان با مسیر غیرمستقیم از راه یک~$w$ نیز به آن
رسید. رابطه‌ای ضعیفاً جهت‌دار است اگر هرگاه بتوان از جهانی~$w$ به
جهان‌های~$u$ و~$v$ رسید، بتوان از هر دو~$u$ و~$v$ به یک جهان واحد~$t$
رسید؛ این خاصیت گاه «خاصیت لوزی» یا «همگرایی» نامیده می‌شود.

    \begin{tabular}{| p{.50\textwidth} || p{.43\textwidth} |}

      {\emph{اگر~$R$ چنین باشد:}} & {\emph{آنگاه این عبارت در~$\mModel{M}$ صادق است:}} \\
      \hline \hline
      \emph{نیمه‌تابعی}: & 
      \multirow{2}{*}{$\Diamond p \lif \Box p$} \\
      $ \forall w \forall u \forall v ((Rwu \land Rwv) \lif u =v )$ & \\

      \multirow{2}{*}{\emph{تابعی}: $\forall w \exists v \forall u(Rwu \liff \eq[u][v]) $} 
      & \multirow{2}{*}{$\Diamond p \liff \Box p$} \\
      & \\
      \hline
      \emph{ضعیفاً چگال}: &  \multirow{2}{*}{$\Box \Box p \lif
        \Box p$} \\
      $\forall u\forall v(Ruv \lif \exists w(Ruw \land Rwv))$ & \\
      \hline
      \emph{ضعیفاً همبند}: &
      \multirow{3}{*}{\hbox to.43\textwidth{$
        \begin{array}{@{}l@{}}
          \Box ((p \land \Box p) \lif q) \lor {} \\
          \qquad \Box ((q \land \Box q) \lif p)
        \end{array}$ \hfill (\Ax{L})}
      } \\
      $\forall w \forall u \forall v ((Rwu \land Rwv) \lif {}$ &\\
      \qquad $(Ruv \lor u=v \lor Rvu))$ & \\
      \hline 
      \emph{ضعیفاً جهت‌دار}: & \multirow{3}{*}{\hbox to .43\textwidth
        {$\Diamond\Box p \lif \Box\Diamond p$\hfill (\Ax{G})}} \\
      $\forall w \forall u \forall v ((Rwu \land Rwv) \lif {}$ & \\
      \qquad $\exists t (Rut \land Rvt))$ & \\
      \hline
    \end{tabular}
    \caption{پنج واقعیت تناظر دیگر.}
    \ollabel{tab:anotherfive}
  \end{table} 

\begin{prob}
  فرض کنید~$\mModel{M} = \tuple{W, R, V}$ یک مدل باشد. نشان دهید اگر
  $R$ خواص سمت چپِ \olref[nml][frd][acc]{tab:anotherfive} را برآورده
  کند، هر نمونهٔ فرمول متناظر در سمت راست در~$\mModel{M}$ صادق است.
\end{prob}
