\olsection{تعریف‌پذیری قاب}

اگرچه عکس استلزام‌های~\olref[acc]{thm:soundschemas} برقرار نیست، با
جایگزین‌کردن «مدل» با «قاب» برقرار می‌شود: برای خواصی که در
\olref[acc]{thm:soundschemas} بررسی شدند، اگر !!a{formula} در یک
\emph{قاب} معتبر باشد، رابطهٔ دسترسی آن قاب واقعاً خاصیت متناظر را
دارد. بنابراین !!{formula}های بررسی‌شده رده‌های قاب‌های دارای خاصیت
متناظر را \emph{تعریف می‌کنند}.

\begin{defn}
  اگر~$\mClass{F}$ رده‌ای از قاب‌ها باشد، می‌گوییم~$!A$ ردهٔ
  $\mClass{F}$ را \emph{تعریف می‌کند} اگر و تنها اگر
  $\mModel{F} \Entails !A$ دقیقاً برای همهٔ قاب‌های
  $\mModel{F} \in \mClass{F}$.
\end{defn}

اکنون به اثبات نتایج کامل تعریف‌پذیری برای قاب‌ها می‌پردازیم.

\begin{thm}\ollabel{thm:fullCorrespondence}
اگر !!{formula} سمت راستِ \olref[acc]{tab:five} در قاب~$\mModel{F}$
معتبر باشد، آنگاه~$\mModel{F}$ خاصیت سمت چپ را دارد.
\end{thm}

\begin{proof}
  \begin{enumerate}
  \item فرض کنید~\Ax{D} در~$\mModel{F} = \tuple{W, R}$ معتبر باشد؛
    یعنی~$\mModel{F} \Entails \Box p \lif \Diamond p$. بگذارید
    $\mModel{M} = \tuple{W, R, V}$ مدلی مبتنی بر~$\mModel{F}$ و
    $w \in W$ باشد. باید نشان دهیم~$v$ای وجود دارد چنان‌که~$Rwv$.
    خلاف آن را فرض کنید: آنگاه برای هر~$!A$، از جمله~$p$، هم
    $\mSat{M}{\Box !A}[w]$ و هم~$\mSat/{M}{\Diamond !A}[w]$. اما در
    این صورت~$\mSat/{M}{\Box p \lif \Diamond p}[w]$، که با فرض
    $\mModel{F} \Entails \Box p \lif \Diamond p$ متناقض است.
  \item فرض کنید~\Ax{T} در~$\mModel{F}$ معتبر باشد؛ یعنی
    $\mModel{F} \Entails \Box p \lif p$. جهان دلخواه~$w \in W$ را
    بگیرید؛ باید~$Rww$ را نشان دهیم. قرار دهید~$u \in V(p)$ اگر و تنها
    اگر~$Rwu$ (اگر~$q$ غیر از~$p$ باشد، $V(q)$ دلخواه است، مثلاً
    $V(q) = \emptyset$). بگذارید~$\mModel{M} = \tuple{W, R, V}$. بنا
    بر تعریف~$V$، برای هر~$u$ چنان‌که~$Rwu$، $\mSat{M}{p}[u]$؛ پس
    $\mSat{M}{\Box p}[w]$. اما بنا بر فرض، $\Box p \lif p$ در~$w$
    صادق است؛ بنابراین~$\mSat{M}{p}[w]$، و بنا بر تعریف~$V$ این تنها
    هنگامی ممکن است که~$Rww$.
  \item نقیض حکم را اثبات می‌کنیم: فرض کنید~$\mModel{F}$ متقارن نباشد؛
    نشان می‌دهیم~\Ax{B}، یعنی~$p \lif \Box\Diamond p$، در
    $\mModel{F}= \tuple{W, R}$ معتبر نیست. اگر~$\mModel{F}$ متقارن
    نباشد، $u,v \in W$هایی وجود دارند چنان‌که~$Ruv$ اما نه~$Rvu$.
    $V$ را چنان تعریف کنید که~$w \in V(p)$ اگر و تنها اگر~$Rvw$
    برقرار نباشد (و~$V$ در موارد دیگر دلخواه باشد). بگذارید
    $\mModel{M} =\tuple{W, R,V}$. بنا بر تعریف~$V$، برای هر~$w$ که
    $Rvw$ برقرار نیست، $\mSat{M}{p}[w]$؛ به‌ویژه چون~$Rvu$ برقرار
    نیست، $\mSat{M}{p}[u]$. همچنین چون~$Rvw$ اگر و تنها اگر
    $w \notin V(p)$، هیچ~$w$ای وجود ندارد چنان‌که هم~$Rvw$ و هم
    $\mSat{M}{p}[w]$؛ پس~$\mSat/{M}{\Diamond p}[v]$. چون~$Ruv$،
    $\mSat/{M}{\Box\Diamond p}[u]$ نیز برقرار است. در نتیجه
    $\mSat/{M}{p \lif \Box\Diamond p}[u]$، و~\Ax{B} در~$\mModel{F}$
    معتبر نیست.
  \item فرض کنید~\Ax{4} در~$\mModel{F} = \tuple{W,R}$ معتبر باشد؛ یعنی
    $\mModel{F} \Entails \Box p \lif \Box\Box p$؛ و جهان‌های دلخواه
    $u,v,w \in W$ را چنان بگیرید که~$Ruv$ و~$Rvw$. باید~$Ruw$ را نشان
    دهیم. $V$ را چنان تعریف کنید که~$z \in V(p)$ اگر و تنها اگر~$Ruz$
    (و~$V$ در موارد دیگر دلخواه باشد). بگذارید
    $\mModel{M} =\tuple{W, R,V}$. بنا بر تعریف~$V$، برای هر~$z$ که
    $Ruz$، $\mSat{M}{p}[z]$؛ بنابراین~$\mSat{M}{\Box p}[u]$. اما بنا
    بر فرض، \Ax{4}، یعنی~$\Box p \lif \Box \Box p$، در~$u$ صادق است؛
    پس~$\mSat{M}{\Box \Box p}[u]$. چون~$Ruv$ و~$Rvw$،
    $\mSat{M}{p}[w]$؛ اما بنا بر تعریف~$V$ این تنها هنگامی ممکن است که
    $Ruw$، چنان‌که می‌خواستیم.
  \item به شیوهٔ نقیض پیش می‌رویم: فرض می‌کنیم قاب
    $\mModel{F} = \tuple{W, R}$ اقلیدسی نیست و نشان می‌دهیم~\Ax{5} را
    کاذب می‌سازد؛ یعنی~$\mModel{F} \Entails/ \Diamond p \lif
      \Box\Diamond p$. فرض کنید جهان‌های~$u,v,w \in W$ وجود دارند
    چنان‌که~$Rwu$ و~$Rwv$، اما~$Ruv$ برقرار نیست. $V$ را چنان تعریف
    کنید که برای هر جهان~$z$، $z \in V(p)$ اگر و تنها اگر~$Ruz$
    برقرار \emph{نباشد}. بگذارید~$\mModel{M} =\tuple{W, R,V}$. آنگاه
    بنا بر فرض~$\mSat{M}{p}[v]$؛ و چون~$Rwv$،
    $\mSat{M}{\Diamond p}[w]$. بااین‌حال، هیچ جهان~$y$ای وجود ندارد که
    هم~$Ruy$ و هم~$\mSat{M}{p}[y]$؛ پس~$\mSat/{M}{\Diamond p}[u]$.
    چون~$Rwu$، نتیجه می‌شود~$\mSat/{M}{\Box\Diamond p}[w]$؛ بنابراین
    \Ax{5}، یعنی~$\Diamond p \lif \Box\Diamond p$، در~$w$ شکست
    می‌خورد.
  \end{enumerate}
\end{proof}

تفاوتی میان اثبات~\Ax{D} و حالت‌های دیگر مشاهده می‌کنید: در آن هیچ
اشاره‌ای به ارزش‌گذاری~$V$ نشد. در واقع، ثابت کردیم اگر
$\mSat{M}{\Box p \lif \Diamond p}$، آنگاه~$\mModel{M}$ مسلسل است. پس
\Ax{D} ردهٔ \emph{مدل‌های} مسلسل را تعریف می‌کند، نه صرفاً قاب‌ها را.

\begin{cor}\ollabel{prop:D-serial}
  هر مدلی که~\Ax{D} در آن صادق باشد مسلسل است.
\end{cor}

\begin{cor}
هر !!{formula} سمت راستِ \olref[acc]{tab:five} ردهٔ قاب‌هایی را تعریف
می‌کند که خاصیت سمت چپ را دارند.
\end{cor}

\begin{proof}
  در~\olref[acc]{thm:soundschemas} ثابت کردیم اگر مدلی خاصیت سمت چپ را
  داشته باشد، !!{formula} سمت راست در آن صادق است. پس اگر قاب
  $\mModel{F}$ خاصیت سمت چپ را داشته باشد، !!{formula} سمت راست در
  $\mModel{F}$ معتبر است. در~\olref{thm:fullCorrespondence} عکس این
  استلزام‌ها را ثابت کردیم: اگر !!a{formula} سمت راست در~$\mModel{F}$
  معتبر باشد، $\mModel{F}$ خاصیت سمت چپ را دارد.
\end{proof}

\begin{prob}
نشان دهید اگر !!{formula} سمت راستِ
\olref[nml][frd][acc]{tab:anotherfive} در قاب~$\mModel{F}$ معتبر باشد،
$\mModel{F}$ خاصیت سمت چپ را دارد. برای این کار، قابی را در نظر بگیرید
که خاصیت سمت چپ را \emph{برآورده نمی‌کند} و~$V$ مناسبی تعریف کنید
چنان‌که !!{formula} سمت راست در یک جهان کاذب باشد.
\end{prob}

\olref{thm:fullCorrespondence} همچنین نشان می‌دهد خواص را می‌توان
ترکیب کرد: برای نمونه، اگر هم~\Ax{B} و هم~\Ax{4} در~$\mModel{F}$ معتبر
باشند، قاب هم متقارن و هم متعدی است، و مانند آن. بسیاری از منطق‌های
موجهات مهم به‌صورت مجموعهٔ !!{formula}های معتبر در همهٔ قاب‌هایی مشخص
می‌شوند که چند خاصیت قاب را ترکیب می‌کنند؛ بنابراین می‌توانیم آن‌ها را
مجموعهٔ !!{formula}های معتبر در همهٔ قاب‌هایی مشخص کنیم که
!!{formula}های تعریف‌کنندهٔ متناظر در آن‌ها معتبرند. برای نمونه، دستگاه
کلاسیک~\Log{S4} مجموعهٔ همهٔ !!{formula}های معتبر در همهٔ قاب‌های
بازتابی و متعدی است، یعنی همهٔ قاب‌هایی که هم~\Ax{T} و هم~\Ax{4} در
آن‌ها معتبرند. \Log{S5} مجموعهٔ همهٔ فرمول‌های معتبر در همهٔ قاب‌های
بازتابی، متقارن و اقلیدسی است، یعنی همهٔ قاب‌هایی که~\Ax{T}، \Ax{B} و
\Ax{5} همگی در آن‌ها معتبرند.

روابط منطقی میان خواص~$R$ عموماً با روابط میان !!{formula}های
تعریف‌کنندهٔ متناظر مطابقت دارند. برای نمونه، هر رابطهٔ بازتابی مسلسل
است؛ پس هرگاه~\Ax{T} در یک قاب معتبر باشد، \Ax{D} نیز معتبر است. (توجه
کنید که این رابطه، رابطهٔ \emph{استلزام} نیست. درست نیست که هرگاه
$\mSat{M}{\Ax{T}}[w]$، آنگاه~$\mSat{M}{\Ax{D}}[w]$.) برخی از این روابط
را ثبت می‌کنیم.

\begin{prop}\ollabel{prop:relation-facts}
  فرض کنید~$R$ رابطه‌ای دوتایی روی مجموعهٔ~$W$ باشد. آنگاه:
  \begin{enumerate}
  \item اگر~$R$ بازتابی باشد، مسلسل است.
  \item اگر~$R$ متقارن باشد، متعدی است اگر و تنها اگر اقلیدسی باشد.
  \item اگر~$R$ متقارن یا اقلیدسی باشد، ضعیفاً جهت‌دار است (یعنی
    «خاصیت لوزی» دارد).
  \item اگر~$R$ اقلیدسی باشد، ضعیفاً همبند است.
  \item اگر~$R$ تابعی باشد، مسلسل است.
  \end{enumerate}
\end{prop}

\begin{prob}
  \olref[nml][frd][def]{prop:relation-facts} را اثبات کنید.
\end{prob}
