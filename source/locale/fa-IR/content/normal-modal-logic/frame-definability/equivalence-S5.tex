\olsection{روابط هم‌ارزی و~\Log{S5}}

منطق موجهات~\Log{S5} به‌صورت مجموعهٔ !!{formula}های معتبر روی همهٔ
قاب‌های کلی مشخص می‌شود؛ یعنی قاب‌هایی که در آن‌ها هر جهان از هر جهان،
از جمله خودش، دسترس‌پذیر است. در چنین وضعی، $\Box$ متناظر با ضرورت و
$\Diamond$ متناظر با امکان است: $\Box !A$ هنگامی صادق است که~$!A$ در
\emph{هر} جهان صادق باشد، و~$\Diamond !A$ هنگامی صادق است که~$!A$ در
\emph{یک} جهان صادق باشد. روشن می‌شود که~\Log{S5} را همچنین می‌توان
به‌صورت !!{formula}های معتبر روی همهٔ قاب‌های بازتابی، متقارن و متعدی،
یعنی روی همهٔ \emph{روابط هم‌ارزی}، مشخص کرد.

\begin{defn}
  رابطهٔ دوتایی~$R$ روی~$W$ یک \emph{رابطهٔ هم‌ارزی} است اگر و تنها
  اگر بازتابی، متقارن و متعدی باشد. رابطهٔ~$R$ روی~$W$ \emph{کلی} است
  اگر و تنها اگر برای همهٔ~$u,v \in W$، $Ruv$.
\end{defn}

چون~\Ax{T}، \Ax{B} و~\Ax{4} به‌ترتیب قاب‌های بازتابی، متقارن و متعدی
را مشخص می‌کنند، قاب‌هایی که رابطهٔ دسترسی در آن‌ها یک رابطهٔ هم‌ارزی
است دقیقاً قاب‌هایی‌اند که هر سه !!{formula} در آن‌ها معتبرند. روشن
می‌شود که روابط هم‌ارزی را می‌توان با ترکیب‌های دیگری از !!{formula}ها
نیز مشخص کرد، زیرا شرایطی که روابط هم‌ارزی را با آن‌ها تعریف کرده‌ایم
با ترکیب‌هایی از شرایط آشنای دیگر بر~$R$ هم‌ارزند.

\begin{prop}\ollabel{prop:equivalences}
  شرایط زیر هم‌ارزند:
  \begin{enumerate}
  \item $R$ یک رابطهٔ هم‌ارزی است؛
  \item $R$ بازتابی و اقلیدسی است؛
  \item $R$ مسلسل، متقارن و اقلیدسی است؛
  \item $R$ مسلسل، متقارن و متعدی است.
  \end{enumerate}
\end{prop}

\begin{proof}
  تمرین.
\end{proof}

\begin{prob}
  \olref[nml][frd][es5]{prop:equivalences} را با نشان‌دادن موارد زیر
  اثبات کنید:
  \begin{enumerate}
  \item اگر~$R$ متقارن و متعدی باشد، اقلیدسی است.
  \item اگر~$R$ بازتابی باشد، مسلسل است.
  \item اگر~$R$ بازتابی و اقلیدسی باشد، متقارن است.
  \item اگر~$R$ متقارن و اقلیدسی باشد، متعدی است.
  \item اگر~$R$ مسلسل، متقارن و متعدی باشد، بازتابی است.
  \end{enumerate}
  توضیح دهید چرا این موارد برای اثبات هم‌ارزی شرایط کافی‌اند.
\end{prob}

\olref{prop:equivalences} همتای معناشناختیِ
\olref[prf][prs]{prop:S5} است، زیرا توصیفی هم‌ارز از منطق موجهاتی قاب‌هایی
به دست می‌دهد که~$R$ روی آن‌ها رابطهٔ هم‌ارزی است (منطقی که به‌طور سنتی
\Log{S5} نامیده می‌شود).

رابطهٔ میان روابط کلی و روابط هم‌ارزی چیست؟ گرچه هر رابطهٔ کلی یک
رابطهٔ هم‌ارزی است، آشکارا هر رابطهٔ هم‌ارزی کلی نیست. بااین‌حال،
!!{formula}های معتبر روی همهٔ روابط کلی دقیقاً همان فرمول‌های معتبر
روی همهٔ روابط هم‌ارزی‌اند.

\begin{prop}
  فرض کنید~$R$ رابطه‌ای هم‌ارزی باشد و برای هر~$w \in W$،
  \emph{ردهٔ هم‌ارزی}~$w$ را مجموعهٔ
  $[w] = \{w'\in W : Rww'\}$ تعریف کنید. آنگاه:
  \begin{enumerate}
  \item $w \in [w]$؛
  \item $R$ روی هر ردهٔ هم‌ارزی~$[w]$ کلی است؛
  \item مجموعهٔ رده‌های هم‌ارزی، $W$ را به زیرمجموعه‌هایی دوبه‌دو
    ناسازگار و در مجموع فراگیر افراز می‌کند.
  \end{enumerate}
\end{prop}

\begin{prop}\ollabel{prop:S5=univ}
  !!^a{formula}~$!A$ در همهٔ قاب‌های~$\mModel{F} =\tuple{W,R}$ که در
  آن‌ها~$R$ رابطه‌ای هم‌ارزی است معتبر است، اگر و تنها اگر در همهٔ
  قاب‌های~$\mModel{F} =\tuple{W,R}$ که در آن‌ها~$R$ کلی است معتبر باشد.
  بنابراین منطق قاب‌های کلی همان~\Log{S5} است.
\end{prop}

\begin{proof}
  به‌آسانی می‌توان بررسی کرد که رابطهٔ کلی~$R$ روی~$W$ رابطه‌ای هم‌ارزی
  است. پس اگر~$!A$ در همهٔ قاب‌هایی که~$R$ در آن‌ها هم‌ارزی است معتبر
  باشد، در همهٔ قاب‌های کلی نیز معتبر است. برای جهت دیگر، به نقیض
  استدلال می‌کنیم: فرض کنید~$!B$ !!a{formula} باشد که در جهان~$w$ از
  مدل~$\mModel{M} = \tuple{W, R,V}$ مبتنی بر قاب~$\tuple{W,R}$ شکست
  می‌خورد، و~$R$ روی~$W$ رابطه‌ای هم‌ارزی باشد. پس
  $\mSat/{M}{!B}[w]$. مدل~$\mModel{M}' = \tuple{W',R',V'}$ را چنین
  تعریف کنید:
  \begin{enumerate}
  \item $W' = [w]$؛
  \item $R'$ روی~$W'$ کلی است؛
  \item $V'(p) = V(p) \cap W'$.
  \end{enumerate}
  (پس مجموعهٔ جهان‌های~$W'$ در~$\mModel{M}'$ با ناحیهٔ سایه‌خورده در
  \olref{fig:partition} نمایش داده شده است.) به‌آسانی می‌توان دید که
  $R$ و~$R'$ روی~$W'$ یکسان‌اند. سپس با استقرا بر !!{formula} می‌توان
  نشان داد که برای هر~$w' \in W'$ و هر~$!A$،
  $\mSat{M'}{!A}[w']$ اگر و تنها اگر~$\mSat{M}{!A}[w']$ (این گزاره
  معنادار است، زیرا~$W' \subseteq W$). به‌ویژه
  $\mSat/{M'}{!B}[w]$؛ پس~$!B$ در مدلی مبتنی بر یک قاب کلی شکست
  می‌خورد.
\end{proof}
