\olsection{سازگاری}

سازگاری خاصیتی مهم برای مجموعه‌های !!{formula} است. مجموعه‌ای از
!!{formula}ها ناسازگار است اگر تناقضی مانند~$\lfalse$ از آن
!!{derivable} باشد، و در غیر این صورت سازگار است. اگر مجموعه‌ای
ناسازگار باشد، همهٔ !!{formula}های آن نمی‌توانند در جهانی از یک مدل
صادق باشند. برای قضیهٔ تمامیت، عکس این حکم را ثابت می‌کنیم: هر مجموعهٔ
سازگار در جهانی از یک مدل، یعنی در «مدل کانونی»، صادق است.

\begin{defn}
  مجموعهٔ~$\Gamma$ نسبت به دستگاه~$\Sigma$ \emph{سازگار}، یا به تعبیر
  ما $\Sigma$-سازگار، است اگر و تنها اگر
  $\Gamma \Proves/[\Sigma] \lfalse$.
\end{defn}

برای نمونه، مجموعهٔ~$\{ \Box(p \lif q), \Box p, \lnot\Box q \}$ نسبت
به منطق گزاره‌ای سازگار است، اما~\Log{K}-سازگار نیست. به همین ترتیب،
مجموعهٔ~$\{ \Diamond p, \Box\Diamond p \lif q, \lnot q \}$
\Log{K5}-سازگار نیست.

\begin{prop}\ollabel{prop:consistencyfacts}
  فرض کنید~$\Gamma$ مجموعه‌ای از !!{formula} باشد. آنگاه:
  \begin{enumerate}
  \item $\Gamma$، $\Sigma$-سازگار است اگر و تنها اگر !!{formula}ای
    چون~$!A$ وجود داشته باشد چنان‌که~$\Gamma \Proves/[\Sigma] !A$.
  \item \ollabel{prop:consistencyfacts-b}%
    $\Gamma \Proves[\Sigma] !A$ اگر و تنها اگر
    $\Gamma \cup \{\lnot!A\}$، $\Sigma$-سازگار نباشد.
  \item \ollabel{prop:consistencyfacts-c}%
    اگر~$\Gamma$، $\Sigma$-سازگار باشد، آنگاه برای هر !!{formula}~$!A$،
    یا~$\Gamma \cup \{!A\}$، $\Sigma$-سازگار است یا
    $\Gamma \cup \{\lnot!A\}$، $\Sigma$-سازگار است.
  \end{enumerate}
\end{prop}

\begin{proof}
  این واقعیت‌ها به‌آسانی با منطق گزاره‌ای کلاسیک نتیجه می‌شوند. استدلال
  مربوط به~\olref{prop:consistencyfacts-c} را ارائه می‌کنیم. به نقیض
  پیش بروید و فرض کنید نه~$\Gamma \cup \{!A\}$ و نه
  $\Gamma \cup \{\lnot!A\}$، $\Sigma$-سازگار است. آنگاه بنا بر
  \olref{prop:consistencyfacts-b}، هم
  $\Gamma, !A \Proves[\Sigma] \lfalse$ و هم
  $\Gamma, \lnot !A \Proves[\Sigma] \lfalse$. بنا بر قضیهٔ استنتاج،
  $\Gamma \Proves[\Sigma] !A \lif \lfalse$ و
  $\Gamma \Proves[\Sigma] \lnot!A \lif \lfalse$. اما
  $(!A \lif \lfalse) \lif ((\lnot!A \lif \lfalse) \lif \lfalse)$
  یک نمونهٔ همان‌گویی است؛ پس بنا بر
  \olref[prp]{prop:derivabilityfacts}\olref[prp]{prop:derivabilityfacts-ruleT}،
  $\Gamma \Proves[\Sigma] \lfalse$.
\end{proof}
