\olsection{کاهش~$\beta$}

\begin{lem}\ollabel{lem:one-par}
  اگر~$M \bredone M'$، آنگاه~$M \bredpar M'$.
\end{lem}
\begin{proof} اگر~$M \bredone M'$، آنگاه برای متغیر~$x$ و ترم‌های
    $N$ و~$Q$، $M$ برابر~$(\lambd[x][N])Q$ و~$M'$ برابر
    $\Subst{N}{Q}{x}$ است. چون بنا بر~\olref[pb]{thm:refl} داریم
    $N \bredpar N$ و~$Q \bredpar Q$، از
    \olref[pb]{defn:bredpar}\olref[pb]{defn:bredpar4} بی‌درنگ
    $(\lambd[x][N])Q \bredpar \Subst{N}{Q}{x}$ به دست می‌آید.
\end{proof}

\begin{lem}\ollabel{lem:par-red}
  اگر~$M \bredpar M'$، آنگاه~$M \bred M'$.
\end{lem}

\begin{proof} با استقرا بر~!!{derivation}ِ $M \bredpar M'$.
  \begin{enumerate}
  \item قاعدهٔ آخر~\olref[pb]{defn:bredpar1} است: در این صورت~$M$ و
    $M'$ هر دو همان~$x$ هستند، و~$x \bred x$.
  \item قاعدهٔ آخر~\olref[pb]{defn:bredpar2} است: $M$ برابر
    $\lambd[x][N]$ و~$M'$ برابر~$\lambd[x][N']$ است، برای~$x$، $N$ و
    $N'$ی که~$N \bredpar N'$. بنا بر فرض استقرا، $N \bred N'$. پس
    $\lambd[x][N] \bred \lambd[x][N']$ (با همان رشتهٔ انقباض‌های
    $\bredone$ که~$N \bred N'$ را نشان می‌دهد).
  \item قاعدهٔ آخر~\olref[pb]{defn:bredpar3} است: $M$ برابر~$PQ$ و
    $M'$ برابر~$P'Q'$ است، برای~$P$، $Q$، $P'$ و~$Q'$ی که
    $P \bredpar P'$ و~$Q \bredpar Q'$. بنا بر فرض استقرا
    $P \bred P'$ و~$Q \bred Q'$. بنابراین~$PQ \bred P'Q'$، با دنبالهٔ
    کاهش~$P \bred P'$ و سپس کاهش~$Q \bred Q'$.
  \item قاعدهٔ آخر~\olref[pb]{defn:bredpar4} است: $M$ برابر
    $(\lambd[x][N])Q$ و~$M'$ برابر~$\Subst{N'}{Q'}{x}$ است، برای
    $x$، $N$، $N'$، $Q$ و~$Q'$ی که~$N \bredpar N'$ و~$Q \bredpar Q'$.
    بنا بر فرض استقرا، $Q \bred Q'$ و~$N \bred N'$. پس
    $(\lambd[x][N])Q \bred \Subst{N'}{Q'}{x}$، با~$N \bred N'$ و سپس
    $Q \bred Q'$ و سرانجام انقباض~$(\lambd[x][N'])Q'$ به
    $\Subst{N'}{Q'}{x}$.
  \end{enumerate}
\end{proof}

\begin{lem}\ollabel{lem:str}
  رابطهٔ~$\bred$ کوچک‌ترین رابطهٔ تعدیِ شامل~$\bredpar$ است.
\end{lem}

\begin{proof}
  بگذارید~$\xred$ کوچک‌ترین رابطهٔ تعدیِ شامل~$\bredpar$ باشد.

  $\bred \subseteq \xred$: فرض کنید~$M \bred M'$؛ یعنی
  $M \ident M_1 \bredone \dots \bredone M_k \ident M'$. بنا بر
  \olref{lem:one-par}، داریم
  $M \ident M_1 \bredpar \dots \bredpar M_k \ident M'$. چون~$\xred$
  شامل~$\bredpar$ و تعدی است، $M \xred M'$.

  $\xred \subseteq \bred$: فرض کنید~$M \xred M'$؛ یعنی
  $M \ident M_1 \bredpar \dots \bredpar M_k \ident M'$. بنا بر
  \olref{lem:par-red}، داریم
  $M \ident M_1 \bred \dots \bred M_k \ident M'$. چون~$\bred$ تعدی
  است، $M \bred M'$.
\end{proof}

\begin{thm}\ollabel{thm:cr}
  رابطهٔ~$\bred$ خاصیت چرچ--راسر را دارد.
\end{thm}

\begin{proof}
  بی‌درنگ از~\olref[dap]{thm:str}، \olref[pb]{thm:cr} و
  \olref{lem:str}.
\end{proof}
