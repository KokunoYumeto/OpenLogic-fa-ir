\olsection{کاهش~$\beta\eta$}

خاصیت چرچ--راسر برای کاهش~$\beta\eta$ ($\bered$) برقرار است.

\begin{lem}\ollabel{lem:one-par}
  اگر~$M \beredone M'$، آنگاه~$M \beredpar M'$.
\end{lem}

\begin{proof} 
  با استقرا بر~!!{derivation}ِ $M \beredone M'$. اگر~$M \beredone M'$
  با تبدیل~$\eta$ باشد (یعنی~\olref[syn][eta]{defn:beredone})، از
  \olref[pbe]{defn:beredpar5} همراه با~\olref[pbe]{thm:refl} استفاده
  می‌کنیم. دیگر حالت‌ها مانند~\olref[b]{lem:one-par} هستند.
\end{proof}


\begin{lem}\ollabel{lem:par-red}
  اگر~$M \beredpar M'$، آنگاه~$M \bered M'$.
\end{lem}

\begin{proof} با استقرا بر~!!{derivation}ِ $M \beredpar M'$.

  چهار حالت نخست مانند حالت‌های~\olref[b]{lem:par-red} هستند.
  اگر قاعدهٔ آخر~\olref[pbe]{defn:beredpar5} باشد، آنگاه~$M$ برابر
  $\lambd[x][Nx]$ و~$M'$ برابر~$N'$ است، برای~$x$، $N$ و~$N'$ی که
  $x \notin \FV{N}$ و~$N \beredpar N'$. بنابراین می‌توانیم نخست
  $\lambd[x][Nx]$ را با تبدیل~$\eta$ به~$N$ کاهش دهیم و سپس رشتهٔ
  گام‌های~$\beredone$ را که~$N \bered N'$ نشان می‌دهد دنبال کنیم؛
  وجود این رشته از فرض استقرا به دست می‌آید.
\end{proof}


\begin{lem}\ollabel{lem:str}
  رابطهٔ~$\bered$ کوچک‌ترین رابطهٔ تعدیِ شامل~$\beredpar$ است.
\end{lem}

\begin{proof}
  همانند~\olref[b]{lem:str}.
\end{proof}

\begin{thm}\ollabel{thm:cr}
  رابطهٔ~$\bered$ خاصیت چرچ--راسر را دارد.
\end{thm}

\begin{proof}
  بنا بر~\olref[dap]{thm:str}، \olref[pbe]{thm:cr} و~\olref{lem:str}.
\end{proof}
