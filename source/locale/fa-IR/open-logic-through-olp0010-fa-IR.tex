% Keep every complete chapter-relative counter in one logical LTR run inside
% Persian prose. Use primitive numeric expansion so Babel does not insert a
% separate directional wrapper around each component (which reverses 1.10 to
% 10.1 even though one-digit pairs can look correct by accident). thmtools
% gives each sibling environment its own presentation macro even though all
% of them advance the shared thm counter, so override every such macro.
\newcommand*{\OLFaChapterCounter}[1]{%
  \babelsublr{\number\value{chapter}.\number\value{#1}}}
\renewcommand*{\thesection}{\OLFaChapterCounter{section}}
\renewcommand*{\thethm}{\OLFaChapterCounter{thm}}
\renewcommand*{\theex}{\OLFaChapterCounter{thm}}
\renewcommand*{\thelem}{\OLFaChapterCounter{thm}}
\renewcommand*{\theprop}{\OLFaChapterCounter{thm}}
\renewcommand*{\thecor}{\OLFaChapterCounter{thm}}
\renewcommand*{\thedefn}{\OLFaChapterCounter{thm}}
\renewcommand*{\theprob}{\OLFaChapterCounter{prob}}
\renewcommand*{\thefigure}{\OLFaChapterCounter{figure}}
