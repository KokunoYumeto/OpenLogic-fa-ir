% Scale only the print title page's fixed skips while the screen title page
% is active; the underlying print driver remains untouched.

  \IfBooleanTF{#1}%
    {\OLScreenOriginalVspace*{\glueexpr#2*1/3\relax}}%

  \let\vfill\OLScreenOriginalVfill

% Track nested subfiles in a wrapper-local \olfilename so exact imported
% basenames can receive narrowly scoped screen-only layout profiles.
\makeatletter
\newcommand*{\OLScreenInstallSubfileTracking}{%
  \let\OLScreenOriginalTable\table
  \let\endOLScreenOriginalTable\endtable
  \let\OLScreenOriginalTabular\tabular
  \let\endOLScreenOriginalTabular\endtabular
  \let\OLScreenOriginalSubfile\subfile
  \RenewDocumentCommand{\subfile}{m}{%
    \begingroup
    \filename@parse{##1}%
    \edef\olfilename{\filename@base}%
    \OLScreenPrepareSubfileLayout
    \OLScreenOriginalSubfile{##1}%
    \endgroup
  }%
}
\makeatother

% The retained closure sequence ends with a heading-only Chapter 8 driver
% immediately followed by the Part II driver.  Memoir normally clears before
% every part; suppress that one redundant clear so the two retained drivers
% form one intentional transition screen.  All other part clears are exact
% memoir behaviour.
\makeatletter
\newcommand*{\OLScreenDefaultPartClear}{%
  \if@openright
    \cleardoublepage
  \else
    \clearpage
  \fi
}
\newcommand*{\OLScreenInstallPartTransition}{%
  \renewcommand{\@setuppart}{%
    \ifnum\value{chapter}=8\relax
      \ifnum\value{part}=1\relax
        \par
      \else
        \OLScreenDefaultPartClear
      \fi
    \else
      \OLScreenDefaultPartClear
    \fi
    \thispagestyle{part}%
    \if@twocolumn
      \onecolumn
      \@tempswatrue
    \else
      \@tempswafalse
    \fi
    \beforepartskip
  }%
}
\makeatother

% Proof trees use \vss internally.  Infinite-order shrink is harmless in the
% print tables, but longtable's page splitter quite correctly refuses it when
% a screen table becomes breakable.  Preserve infinite stretch for vertical
% centering while giving the splitter a large finite shrink component.  The
% environment hook is locally scoped by longtable and covers both the nine
% wrapper-created rule tables and source-native longtables in this supplement.
\newcommand*{\OLScreenFiniteVss}{\vskip0pt plus 1fil minus 10000pt\relax}
\AddToHook{env/longtable/begin}{\let\vss\OLScreenFiniteVss}
\makeatletter
\NewDocumentEnvironment{OLScreenFlowTable}{o}{%
  \par
  \def\@captype{table}%
  \Needspace{6\baselineskip}%
}{\par}
\makeatother
\NewDocumentEnvironment{OLScreenFlowTabular}{O{} m}{%
  \advance\OLScreenTabularDepth by 1
  \ifnum\OLScreenTabularDepth=1
    \begin{longtable}{#2}%
  \else
    \if\relax\detokenize{#1}\relax
      \OLScreenOriginalTabular{#2}%
    \else
      \OLScreenOriginalTabular[#1]{#2}%
    \fi
  \fi
}{%
  \ifnum\OLScreenTabularDepth=1
    \end{longtable}%
  \else
    \endOLScreenOriginalTabular
  \fi
}
