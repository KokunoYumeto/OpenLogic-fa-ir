% The print title page deliberately uses generous fixed skips. Scale only
% those skips while the screen title page is active; the print driver keeps
% its original values and behavior.
\newcommand*{\OLScreenOriginalVspace}{}
\newcommand*{\OLScreenOriginalVfill}{}
\NewDocumentCommand{\OLScreenScaledVspace}{s m}{%
  \IfBooleanTF{#1}%
    {\OLScreenOriginalVspace*{\glueexpr#2*1/3\relax}}%
    {\OLScreenOriginalVspace{\glueexpr#2*1/3\relax}}%
}
\newcommand*{\OLScreenCompactVfill}{\par\vskip.35\baselineskip\small}
\AddToHook{env/titlingpage/begin}{%
  \let\OLScreenOriginalVspace\vspace
  \let\OLScreenOriginalVfill\vfill
  \let\vspace\OLScreenScaledVspace
  \let\vfill\OLScreenCompactVfill
}
\AddToHook{env/titlingpage/end}{%
  \let\vspace\OLScreenOriginalVspace
  \let\vfill\OLScreenOriginalVfill
}

% Formal displays may opt into the 852 pt safe live width without widening
% ordinary prose or nested/list material.
\newlength{\OLScreenWideOverhang}
\setlength{\OLScreenWideOverhang}{54bp}
\newif\ifOLScreenWideActive
\NewDocumentEnvironment{OLScreenWide}{}{%
  \par
  \ifdim\linewidth=\textwidth\relax
    \OLScreenWideActivetrue
    \begin{adjustwidth}{-\OLScreenWideOverhang}{-\OLScreenWideOverhang}%
  \else
    \OLScreenWideActivefalse
    \PackageWarning{open-logic-screen}{Nested wide block kept at current linewidth}%
  \fi
}{%
  \ifOLScreenWideActive\par\end{adjustwidth}\fi
  \par
}

% Track nested subfiles in a wrapper-local \olfilename so exact imported
% basenames can receive narrowly scoped screen-only layout profiles.
\makeatletter
\newcommand*{\OLScreenInstallSubfileTracking}{%
  \let\OLScreenOriginalSubfile\subfile
  \RenewDocumentCommand{\subfile}{m}{%
    \begingroup
    \filename@parse{##1}%
    \edef\olfilename{\filename@base}%
    \OLScreenOriginalSubfile{##1}%
    \endgroup
  }%
}
\makeatother

\newcommand*{\OLScreenUseTableProfile}[2]{%
  \fontsize{#1}{#2}\selectfont
}
\newcommand*{\OLScreenOriginalTabular}{}
\newcount\OLScreenCorrespondenceTableIndex
\NewDocumentEnvironment{OLScreenProfiledTabular}{o m}{%
  \IfNoValueTF{#1}
    {\OLScreenOriginalTabular{| p{.51\textwidth} || p{.44\textwidth} |}}
    {\OLScreenOriginalTabular[#1]{| p{.51\textwidth} || p{.44\textwidth} |}}%
}{%
  \endOLScreenOriginalTabular
}
\newcommand*{\OLScreenApplyCompleteTableProfile}{%
  \OLScreenIfExpandedStringTF{\theolpart}{nml}{%
    \OLScreenIfExpandedStringTF{\theolchapter}{tab}{%
      \OLScreenIfExpandedStringTF{\theolsection}{mru}
        {\OLScreenUseTableProfile{15pt}{18pt}}{}%
    }{%
      \OLScreenIfExpandedStringTF{\theolchapter}{frd}{%
        \OLScreenIfExpandedStringTF{\theolsection}{acc}{%
          \OLScreenUseTableProfile{16pt}{19pt}%
          \global\advance\OLScreenCorrespondenceTableIndex by 1
          \PackageInfo{open-logic-screen}{Correspondence table index
            \the\OLScreenCorrespondenceTableIndex}%
          \ifnum\OLScreenCorrespondenceTableIndex=2
            \setlength{\tabcolsep}{4pt}%
            \PackageInfo{open-logic-screen}{Applying audited .51/.44
              correspondence-table profile}%
            \let\OLScreenOriginalTabular\tabular
            \let\endOLScreenOriginalTabular\endtabular
            \let\tabular\OLScreenProfiledTabular
            \let\endtabular\endOLScreenProfiledTabular
          \fi
        }{}%
      }{}%
    }%
  }{}%
}
\newcommand*{\OLScreenInstallTableWrapper}{%
  \let\OLScreenOriginalTable\table
  \let\endOLScreenOriginalTable\endtable
  \RenewDocumentEnvironment{table}{o}{%
    \IfNoValueTF{##1}
      {\OLScreenOriginalTable}
      {\OLScreenOriginalTable[##1]}%
    \OLScreenApplyCompleteTableProfile
  }{%
    \endOLScreenOriginalTable
  }%
}

% The print macro reserves only .8 em for a signed-formula prefix. At 17 pt,
% indexed signs such as S_i exceed that box; the screen edition gives them
% their natural width instead of accepting an internal overfull box.
\NewDocumentCommand{\OLScreenSignedFormula}{m m o}{%
  \ensuremath{%
    \IfNoValueTF{#3}{}{#3\,}%
    \hbox to1.05em{\ensuremath{#1}\hfil} #2}%
}

    \babelsublr{%
      \IfFileExists{#2}
        {\includegraphics[width=#1]{#2}}
        {\includegraphics[width=#1]{\olpath/#2}}}}%
  \par
}
