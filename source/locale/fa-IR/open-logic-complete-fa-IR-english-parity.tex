% The English source draws diagrams and its one source-supplied sideways
% derivation in an LTR graphic box.  Babel's RTL paragraph direction otherwise
% displaces those boxes by a full line width.  Change direction only while the
% graphic box is assembled; Persian prose and captions retain their own bidi.
\makeatletter
\newcommand*{\OLParityGraphicLTRBefore}{\begingroup\bbl@setdirs\z@}
\newcommand*{\OLParityGraphicLTRAfter}{\endgroup}
\newcommand*{\OLParityInstallGraphicDirectionFixes}{%
  \let\OLParityOriginalAsset\olasset
  \RenewDocumentCommand{\olasset}{O{\olphotowidth}m}{%
    \begingroup\bbl@setdirs\z@
    \OLParityOriginalAsset[##1]{##2}%
    \endgroup
  }%
  \AddToHook{env/tikzpicture/before}{\OLParityGraphicLTRBefore}%
  \AddToHook{env/tikzpicture/after}{\OLParityGraphicLTRAfter}%
  \AddToHook{env/sidewaysfigure/before}{\OLParityGraphicLTRBefore}%
  \AddToHook{env/sidewaysfigure/after}{\OLParityGraphicLTRAfter}%
}
\makeatother
