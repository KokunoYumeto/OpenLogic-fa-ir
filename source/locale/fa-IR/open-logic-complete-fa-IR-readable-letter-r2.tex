\AddToHook{begindocument/before}{%
  % Empirical RTL rule probe: Babel reverses this register's physical motion.
  % +0.8773pt moves the 466.252bp rule from x=73.748bp to x=72.874bp,
  % centering it at x=306.000bp on the 612bp Letter page.
  \addtolength{\oddsidemargin}{0.8773pt}%
  \addtolength{\evensidemargin}{0.8773pt}%
  \OLReadableInstallGraphicDirectionFixes
  % Three source proofs contain unbreakable inline/display formula runs.  Use
  % the smallest local reduction that keeps each proof inside the page; all
  % ordinary Persian prose remains at the 14 pt edition size.
  \AddToHook{env/proof/begin}{%
    \ifnum\value{chapter}=16\relax
      \ifnum\value{section}=5\relax\fontsize{12pt}{15pt}\selectfont\fi
    \fi
    \ifnum\value{chapter}=23\relax
      \ifnum\value{section}=7\relax\fontsize{13pt}{16pt}\selectfont\fi
    \fi
    \ifnum\value{chapter}=43\relax
      \ifnum\value{section}=2\relax\fontsize{13.5pt}{17pt}\selectfont\fi
    \fi
  }%
  \hypersetup{%
    pdfsubject={\OLCompletePDFSubject},%
    pdfkeywords={\OLCompletePDFKeywords},%
    pdfstartview=FitH,%
    pdfpagelayout=OneColumn,%
    pdfpagemode=UseOutlines%
  }%
}
